\section{The gain at a prime}\label{sec:gain}

We use \(q_0\), \(d\), \(K\), \(\chi_K\), \(q_1\), \(E\), \(\Phi\), \(f\), \([\nu]\), \(\Phi_a\), \(\Phi_b\), the weight \(\kappa\) and \(\sfin\) as in Chapter~\ref{da-chap}; recall that \(a/x_1 \le \kappa(\sigma) \le b \le a\) and that \(q_1\) is \(q_0\) times a product of powers of \(2\) and \(3\).  For a natural number \(p\), let \(c_p = 1/2\) if \(p = d\) and \(c_p = 1\) otherwise, and let \(I_p\) be the set of integers \(e \ge 1\) such that \(p = d\), or \(\chi_K(p) = -1\) and \(e\) is odd.  For \(y \in \mathbb R\) and \(\sigma \in \Phi\), with \(z_+ = \max(z, 0)\), the \emph{local prime sum} and the \emph{gain} are
\[
  \sfp(y, \sigma) = c_p \sum_{e \in I_p} \Bigl(\frac{y}{p^e} - \kappa([-p^e]\sigma)\Bigr)_+ ,
  \qquad
  \gain(p) = m\Bigl(\sum_{\tau \in \Phi_a} \sfp(a, \tau) + k \sum_{\sigma \in \Phi} \sfp(\kappa(\sigma), \sigma)\Bigr) .
\]
The first sum records the \(a\) columns of height one at each \(a\)-place; the second counts the local term at \(\kappa(\sigma)\) with multiplicity \(k\) at every place.

\begin{problem}\label{pr:qone-split}
Every prime dividing \(q_1\) splits in \(K\).  Hence a prime \(p \ne d\) with \(\chi_K(p) \ne 1\) is prime to \(q_1\), and \([-p^e]\) is the place of the class of \(-p^e\).
\end{problem}

\begin{problem}\label{pr:shape}
Let \(p \ne d\) be inert in \(K\), \(\tau \in \Phi_a\), and suppose that \(\kappa([-p^e]\tau) = w\) for every positive odd integer \(e\).  Then \(\sfp(a, \tau) = \sum_{e \ge 1,\ e\ \mathrm{odd}} (a/p^e - w)_+\), the first sum of Theorem~\ref{ip-thm:main} without the integer parts.  Evaluated at \(p = 13\), \(a = 13^4\) and \(w = 1\), the sum is \(2208\).
\end{problem}

\begin{problem}\label{pr:gain-zero}
\(\gain(p) = 0\) if \(\chi_K(p) = 1\) or \(p \ge x_1\).
\end{problem}

\begin{problem}\label{pr:part-at-p}
Let \(p\) be prime and \(y \le a\).  The terms of \(\sfin(y, \sigma)\) with \(\nu\) a power of \(p\) add up to \(2\, \sfp(y, \sigma) \log p\).
\end{problem}

Let \(\Pi\) be the set of primes \(p < x_1\) with \(\chi_K(p) \ne 1\).  Summing over primes, \(2\sum_{p \in \Pi} \gain(p)\log p = m\sum_{\tau \in \Phi_a} \sfin(a, \tau) + mk \sum_{\tau \in \Phi} \sfin(\kappa(\tau), \tau)\); the contribution of the powers of \(p\) to the right-hand side is \(2\gain(p)\log p\).

\section{Collisions}\label{sec:collisions}

In this section \(p\) is a prime.  For a set \(\Phi' \subseteq \Phi\), a \emph{collision outside \(\Phi'\)} is a pair \((\tau, e)\) with \(e \in I_p\) and \(\tau, [-p^e]\tau \in \Phi_a \smallsetminus \Phi'\); the \emph{collision sum} is
\[
  C_p(\Phi') = \sum_{(\tau, e)\ \text{collision outside}\ \Phi'} \frac{a}{p^e} .
\]
Let \(o_p\) be the order of the image of \([-p]\) in \((\Z/q_0)^\times\), and \(r_p\) the least integer \(h \ge 0\) with \(m \le p^h\).  The prime \(p\) is \emph{exceptional} if \(\chi_K(p) = -1\), \(p \not\equiv \pm 1 \pmod m\), and \(o_p\) is odd with \(o_p \le r_p\).  The \emph{unit of loss} at \(p\) is
\[
  L_p = f(\log b)^2\, \frac{a}{p} \cdot \begin{cases} m & \text{if } m \mid p - 1 \text{ or } m \mid p + 1,\\ 1 & \text{otherwise.}\end{cases}
\]

\begin{problem}\label{pr:collision-odd}
Let \(\chi_K(p) \ne 1\) and let \((\tau, e)\) be a collision outside some \(\Phi'\).  Then \(p \ne d\), \(e\) is odd, and \(o_p \mid e\); in particular \(o_p\) is odd.
\end{problem}

\begin{problem}\label{pr:o-example}
Let \(q_0 = 18480\) and let \(p\) be a prime not dividing \(q_0\).  If \(o_p\) is odd, then \(p \equiv -1 \pmod{240}\) and \(o_p \in \{1, 3, 5, 15\}\).  For \(d = 138599\), the inert prime \(13\) has \(o_{13} = 20\).  Among the primes inert in \(\Q(\sqrt{-138599})\), the least with \(o_p\) odd is \(479\), with \(o_{479} = 15\), and the least with \(o_p = 1\) is \(55439\).
\end{problem}

\begin{problem}\label{pr:geometric}
Let \(o \ge 0\) be an integer.  If every collision \((\tau, e)\) outside \(\Phi'\) has \(e \ge o\), then \(C_p(\Phi') \le 2fa/p^o\).
\end{problem}

From Chapter~\ref{da-chap} we use \((\log b)^2 \ge 15\) and \(m \ge 2\).

\begin{problem}\label{pr:non-exceptional}
Let \(p < x_1\) be a prime with \(\chi_K(p) \ne 1\) that is not exceptional.  Then \(m\, C_p(\varnothing) \le L_p\).
\end{problem}

\begin{problem}\label{pr:why-mark}
Let \(p\) be exceptional with \(o_p = 1\), and let \(h_0 = \#(\Z/q_0)^\times\).  Then \(\#\Phi_a = f/h_0\) and \(m\, C_p(\varnothing) \ge m f a/(h_0 p)\), which exceeds \(L_p\) when \(m > h_0(\log b)^2\).
\end{problem}

So at an exceptional prime the unit of loss need not pay for the collisions; Section~\ref{sec:exceptional} obtains the valuation bound there on a submodule.

\section{The chain of lifts}\label{sec:chain}

Fix a prime \(p\) and a ring map \(\eta\colon \C \to \C_p\); \(\Oc_\eta\) is the ring of Chapter~\ref{ob-chap}.

\begin{definition}\label{def:gauge}
The \emph{gauge} \(\lVert\omega\rVert_g\) of a Weyl vector \(\omega\) at a point \(g\) of the finite adelic group, and the phrases ``a form has \emph{valuation at least \(c\)}'' and ``a form is \emph{integral}'', at \(p\) and \(\eta\), are those of Definition~\ref{ob-def:gauge}.
\end{definition}

With \(S \supseteq S_\ram\) the source forms and the ramified ones (Chapter~\ref{ob-chap}), let \(S_\eta \subseteq S\) be the \(\Oc_\eta\)-module of integral source forms and \(S_{\ram,\eta} = S_\eta \cap S_\ram\).

The translations \(\mathfrak t = (\mathfrak t_j)_{j \ge 0} \in \Tc\) and the translated vacua \(v_j(\mathfrak t)\) are those of Definition~\ref{ss-def:definite}.

\begin{definition}\label{def:weak}
For a twist \(\vartheta\), a finitely supported \(\Psi\colon \Tc \to \C\) is a \emph{weak chain datum} for \(\vartheta\) if: for every \(\mathfrak t\) in its support and every \(j\), the entries of \(\bar{\pi}_M\mathfrak t_j\) are integral at every finite place not dividing \(N \Nm Q_\vartheta\); every value \(\Psi(\mathfrak t)\) is integral under every ring map \(\C \to \C_\ell\) for every prime \(\ell \nmid \Nm Q_\vartheta\); if \(Q_\vartheta = \varnothing\), then \(\sum_{\mathfrak t} \lvert\sigma(\Psi(\mathfrak t))\rvert \le \exp(x^{60})\) for every automorphism \(\sigma\) of \(\C\); and \(\sum_{\mathfrak t} \Psi(\mathfrak t) \bigotimes_{j < m} v_j(\mathfrak t)\) is fixed by the levels of \(X_0, \dots, X_{m-1}\).
\end{definition}

\begin{definition}\label{def:restricted}
For a tame character \(\xi\) and a finitely supported \(\Psi\colon\Tc\to\C\), the \emph{restricted chain} is \(R_{\xi,\Psi}:=\mathrm{res}\circ\bigl(\sum_{\mathfrak t}\Psi(\mathfrak t)\,\Theta^{\mathfrak t}_{m-1}\cdots\Theta^{\mathfrak t}_0\bigr)\) on \(S\), where \(\Theta^{\mathfrak t}_j\) is the definite lift from \(X_j\) to \(X_{j+1}\) of Definition~\ref{ss-def:definite}, built from \(\xi\) with datum \(v_j(\mathfrak t)\), and \(\mathrm{res}\) is the restriction to \(Y_m\) of Definition~\ref{ss-def:composite}.  We use this map for a twist \(\vartheta\) and a weak chain datum for \(\vartheta\).  In this chapter and in Chapters~\ref{d2-chap}, \ref{d3-chap}, \ref{d5-chap} and~\ref{d6-chap}, the lift from \(X_j\) to \(X_{j+1}\) carries its source index \(j\ge0\).  The chain uses \(0\le j<m\), corresponding to lifts \(1,\dots,m\) of Chapter~\ref{ss-chap}.
\end{definition}

Away from \(N\) the proof uses the following expansion, which we take from it \cite{GL}.

\begin{theorem}\label{thm:plain}
For every twist \(\vartheta\) and every chain datum \(\Psi\) for \(\vartheta\) there is a weak chain datum \(\Psi'\) for \(\vartheta\) with \(B_{\vartheta,\Psi}\varphi = R_{\vartheta,\Psi'}\varphi\) for all \(\varphi \in S\).
\end{theorem}

We use three inputs.  In~(i) and~(ii), let \(\Psi'\) be a weak chain datum for \(\vartheta\) and suppose \(p \nmid \Nm Q_\vartheta\).
\begin{enumerate}
\item[(i)] If \(p \ne N\) and \(p \ne d\), then \(R_{\vartheta,\Psi'}\) maps \(S_\eta\) to integral forms.
\item[(ii)] Let \(p \ne N\).  For every \(\mathfrak t\) in the support of \(\Psi'\), every \(j\ge0\) and every \(\delta\in\Oc_K\) such that \(\delta=\bar\pi_M\) or \(p\nmid M\), the datum \(v_j(\mathfrak t)\) is standard of denominator \(\delta\) (Definition~\ref{d2-def:standard}).  In particular, one may take \(\delta=1\) when \(p\nmid M\).  If \(\rho_0, \rho_1, \dots\) are real numbers such that, for every \(\mathfrak t\) in the support of \(\Psi'\) and every \(j \ge 0\), the datum \(v_j(\mathfrak t)\) multiplies gauges by at most \(p^{-\rho_j}\) (Definition~\ref{d2-def:raises}, for the lift \(j\) built from \(\vartheta\)), then \(R_{\vartheta,\Psi'}\) maps \(S_\eta\) to forms of valuation at least \(\sum_{j<m} \rho_j\).
\item[(iii)] Let \(\Dc_j(\Phi')\), for \(j \ge 0\) and \(\Phi' \subseteq \Phi\), be the stage bound of Definition~\ref{d2-def:stage}.  By Theorem~\ref{d2-thm:stage}: if \(\chi_K(p) \ne 1\), then for every twist, every \(j \ge 0\) and every standard datum \(\varphi\) of denominator \(1\) on \(\operatorname{Hom}(X_j, X_{j+1})\), \(\varphi\) multiplies gauges by at most \(p^{-\Dc_j(\varnothing)}\); and if \(p < x_1\) and \(\chi_K(p) \ne 1\), then for every \(0 \le j < m\) and every \(\Phi' \subseteq \Phi\),
\[
  \Dc_j(\Phi') \ge \frac{\gain(p)}{m} - C_p(\Phi') + c_p\, \#(\Phi_a \cap \Phi')\,\frac{a}{x_1} - \frac{fa(\log b)^2}{mp} .
\]
\end{enumerate}
Inputs~(i) and~(ii) are read from the proof.  Chapter~\ref{d2-chap} proves the first part of~(iii) from statements (II)--(IV) of Section~\ref{d2-sec:kernel} and the bound on jets along products of minors, Theorem~\ref{pf-thm:count}, applied to the jet series of a standard datum.  It proves the second part by comparing the predicted divisibility of the diagram of the lift with the prime sums that define \(\gain(p)\).  The stage bound credits each erased box with \(\gamma_p := c_p\sum_{e \in I_p} p^{-e}\), which is \(p/(p^2 - 1)\) at an inert prime and \(1/(2d - 2)\) at \(p = d\) (Problem~\ref{d1-pr:gamma-sum}).  A prime \(p < x_1\) divides neither \(M\) nor \(N\), since \(p_0, p_1 \ge b^{10}\) and \(N > M\) (Chapter~\ref{da-chap}).

\begin{problem}\label{pr:gauge-chain}
Let \(\mathcal V_0, \dots, \mathcal V_m\) be finite-dimensional \(\C_p\)-spaces with sup norms in fixed bases, let \(\Theta^{\mathfrak t}_j\colon \mathcal V_j \to \mathcal V_{j+1}\) satisfy \(\lVert\Theta^{\mathfrak t}_j u\rVert \le p^{-\rho_j}\lVert u\rVert\) for \(\mathfrak t\) in a finite set, and let \(\Psi(\mathfrak t) \in \C_p\) have \(\lvert\Psi(\mathfrak t)\rvert_p \le 1\).  Then \(\bigl\lVert\sum_{\mathfrak t} \Psi(\mathfrak t)\Theta^{\mathfrak t}_{m-1}\cdots\Theta^{\mathfrak t}_0 u\bigr\rVert \le p^{-(\rho_0 + \dots + \rho_{m-1})}\lVert u\rVert\).
\end{problem}

\begin{problem}\label{pr:collision-chain}
Let \(p < x_1\) be prime with \(\chi_K(p) \ne 1\), \(p \nmid \Nm Q_\vartheta\), and \(\Psi\) a chain datum for \(\vartheta\).  For \(\varphi \in S_\eta\), \(B_{\vartheta,\Psi}\varphi\) has valuation at least \(\gain(p) - L_p - m\, C_p(\varnothing)\).
\end{problem}

Let \(p \nmid \Nm Q_\vartheta\) and let \(\Psi\) be a chain datum for \(\vartheta\).  If \(p < x_1\) is non-split and not exceptional, then by Problems~\ref{pr:non-exceptional} and~\ref{pr:collision-chain}, \(B_{\vartheta,\Psi}\) has valuation at least \(\gain(p) - 2L_p\) on \(S_\eta\).  If \(p \ne N\) and either \(\chi_K(p) = 1\) or \(p \ge x_1\), then \(p \ne d\), as \(d < x_1\) by Chapter~\ref{da-chap}; by~(i), \(B_{\vartheta,\Psi}\) maps \(S_\eta\) to integral forms, and \(\gain(p) = 0\).

\section{Exceptional primes}\label{sec:exceptional}

Let \(p\) be inert in \(K\).  Each place \(\sigma\), represented by the embedding \(\Phi_\sigma\), gives through \(\eta\) a prime \(v_\sigma\) of \(E\) above \(p\); for every positive odd integer \(e\), \(v_{[-p^e]\sigma} = \bar v_\sigma\), the complex conjugate prime (Problem~\ref{d3-pr:decomposition}).  Let \(p < x_1\) be exceptional.  A \emph{mark} is a set \(\Mk\) of primes of \(E\) above \(p\), each of the form \(v_\tau\) with \(\tau \in \Phi_a\), such that \(\bar v \notin \Mk\) for \(v \in \Mk\).  It is \emph{full} if it contains \(v_\tau\) or \(\bar v_\tau\) for each \(\tau \in \Phi_a\); a full mark exists for every exceptional \(p < x_1\) (Problem~\ref{d3-pr:full-mark-exists}).  Let \(\Phi_{\Mk} = \{\sigma \in \Phi : v_\sigma \in \Mk\}\).

\begin{problem}\label{pr:full-mark}
If \(\Mk\) is a full mark, then \(C_p(\Phi_{\Mk}) = 0\).
\end{problem}

For \(v \in \Mk\) let \(a_v = a \cdot \#\{\tau \in \Phi_a : v_\tau = v\}\), let \(T_v\) be the Hecke operator at the split pair \(v, \bar v\) of the double coset of \(\operatorname{diag}(p 1_{r_p}, 1)\) (\(p\) is a uniformizer at \(v\), as \(p\) is unramified in \(E\)), and let \(Y_v = p^{a_v} T_v\) on \(S\).  Let \(Y_{\Mk}\) be the product of the \(Y_v\), \(v \in \Mk\), in a fixed order.  By Theorem~\ref{d3-thm:full-mark}, for a full mark \(\Mk\), a weak chain datum \(\Psi'\) for \(\vartheta\) with \(p \nmid \Nm Q_\vartheta\), and \(\varphi \in S_\eta\), the form \(R_{\vartheta,\Psi'}(Y_{\Mk}^m\varphi)\) has valuation at least \(\gain(p) - 2L_p\).  Let \(j_p = \lceil 2mfa/p \rceil\).

\begin{problem}\label{pr:crude}
Under the hypotheses of Problem~\ref{pr:collision-chain}, \(B_{\vartheta,\Psi}(p^{j_p}\varphi)\) has valuation at least \(\gain(p) - L_p\) for \(\varphi \in S_\eta\).
\end{problem}

\begin{problem}\label{pr:two-pieces}
Let \(p < x_1\) be exceptional, \(\Mk\) a full mark, \(p \nmid \Nm Q_\vartheta\), and \(\Psi\) a chain datum for \(\vartheta\).  On \(S' = Y_{\Mk}^m S_\eta + p^{j_p} S_\eta\), \(B_{\vartheta,\Psi}\) has valuation at least \(\gain(p) - 2L_p\).
\end{problem}

How large is \(S'\) in \(S_\eta\)?  We use the colength of Chapter~\ref{sd-chap} and the phrase ``onto modulo \(p^u\) up to colength \(c\)'' as defined in Chapter~\ref{hn-chap}.  By~\eqref{d4-eq:mark-colength}, for a mark \(\Mk\), \(v \in \Mk\) and every integer \(u \ge 0\), \(Y_v\) is onto \(S_\eta\) modulo \(p^u\) up to colength \(\bigl(12a_v/(mp) + u/(m^2 f)\bigr) D_X\).  Chapter~\ref{d4-chap} deduces this from the theorem on elementary divisors, Theorem~\ref{d4-thm:elementary-divisors}; the theorem and its application at a marked prime are read from the proof there.

\begin{problem}\label{pr:colength-add}
Let \(\Oc\) be a commutative ring, \(p \in \Oc\), \(\mathcal L\) a submodule of an \(\Oc\)-module \(\mathcal W\), and \(Y_1\), \(Y_2\) \(\Oc\)-linear endomorphisms of \(\mathcal W\) that are onto \(\mathcal L\) modulo \(p^u\) up to colengths \(c_1\) and \(c_2\).  Then \(Y_1Y_2\) is onto \(\mathcal L\) modulo \(p^u\) up to colength \(c_1 + c_2\).
\end{problem}

\begin{problem}\label{pr:budget}
For a mark \(\Mk\), \(\sum_{v \in \Mk} a_v \le af\) and \(\#\Mk \le f\), and
\[
  m \sum_{v \in \Mk} \Bigl(\frac{12 a_v}{mp} + \frac{j_p}{m^2 f}\Bigr) \le \frac{14fa + p/m}{p} \le L_p .
\]
\end{problem}

So \(Y_{\Mk}^m\) is onto \(S_\eta\) modulo \(p^{j_p}\) up to colength \(L_p D_X\): there is an integer \(0 \le l \le L_p D_X\) such that the colength of \(S'\) in \(S_\eta\) divides \(p^l\).

\section{The level prime and the sum of the errors}\label{sec:sum}

At the level prime \(N\), which is at least \(x_1\), the gain is \(0\).  Let \(S_N\) be the module of Definition~\ref{d5-def:SN}.  By Theorem~\ref{d5-thm:level} there is an integer \(0 \le l \le mfb\, D_X\) such that the colength of \(S_N\) in \(S_\eta\) divides \(N^l\), and for every twist \(\vartheta\), every chain datum \(\Psi\) for \(\vartheta\) and every \(\varphi \in S_\ram \cap S_N\), the form \(B_{\vartheta,\Psi}\varphi\) has valuation at least \(-2mfb\).  The module \(S_N\) is defined in the space of forms on \(X_0\).  At \(p=N\) the proof of Theorem~\ref{thm:one-prime} takes its inverse image under the inclusion of the source forms, so \(S'=S_N\cap S\).

The \emph{allowed loss} is
\[
  \loss(p) = \begin{cases} 2L_p & \text{if } p < x_1 \text{ is prime and } \chi_K(p) \ne 1,\\ 2mfb & \text{if } p = N,\\ 0 & \text{otherwise.} \end{cases}
\]

\begin{problem}\label{pr:pm-one}
Let \(m \ge 2\), \(p \ge 2\), and \(p \equiv \pm 1 \pmod m\).  Then \(p = mc - 1\) or \(p = mc + 1\) for an integer \(c\) with \(1 \le c \le p\) and \(m \le 2p/c\).
\end{problem}

Let \(T_0 = 4fa(\log b)^2\log x_1\).

\begin{problem}\label{pr:term}
For a prime \(p < x_1\) with \(\chi_K(p) \ne 1\) there is an integer \(1 \le c \le p\) with \(p \in \{mc - 1, c, mc + 1\}\) and \(\loss(p)\log p \le T_0/c\).
\end{problem}

The term of \(N\) is also at most \(T_0\): the scale inequalities of Chapter~\ref{da-chap} give \(2mfb\log N \le T_0\) (they give \(\log N \le 30\, m\log b\) and \(m \le 2\sqrt{x}\), and imply \((\log b)\log x_1 \ge 60\)).  In sums over natural numbers we set \(\log 0 = 0\).

\begin{problem}\label{pr:harmonic}
For every finite set \(A\) of natural numbers, \(\sum_{p \in A} \loss(p)\log p \le 4T_0(1 + \log x_1)\).
\end{problem}

\begin{problem}\label{pr:final}
If \((\log x_1)^6 \le \log b\) and \(\log x_1 \ge 3\), then \(4T_0(1 + \log x_1) \le U\).
\end{problem}

Both hypotheses of Problem~\ref{pr:final} hold for the data of Chapter~\ref{da-chap}.

\section{The theorem}\label{sec:theorem}

The four kinds of primes are: \(p \ne N\) split or at least \(x_1\), with \(S_\eta\) and \(l = 0\) (Section~\ref{sec:chain}); \(p < x_1\) non-split and not exceptional, with \(S_\eta\) and \(l = 0\) (Section~\ref{sec:chain}); \(p < x_1\) exceptional, with \(S'\) (Section~\ref{sec:exceptional}); and \(p = N\), with \(S' = S_N \cap S\) and the integer \(l\) of Theorem~\ref{d5-thm:level} (Section~\ref{sec:sum}).  These four cases give the following theorem for ramified inputs.

\begin{theorem}\label{thm:one-prime}
Let \(p\) be prime and \(\eta\colon \C \to \C_p\) a ring map.  There are an integer \(l \ge 0\) with \(l \le \loss(p)\, D_X\) and an \(\Oc_\eta\)-submodule \(S' \subseteq S\) whose colength in \(S_\eta\) divides \(p^l\), such that, for every twist \(\vartheta\), every chain datum \(\Psi\) for \(\vartheta\) with \(p \nmid \Nm Q_\vartheta\), and every \(\varphi \in S' \cap S_\ram\), the form \(B_{\vartheta,\Psi}\varphi\) has valuation at least \(\gain(p) - \loss(p)\).
\end{theorem}

Intersecting \(S'\) with \(S_{\ram,\eta}\) keeps the bound on its colength, now in \(S_{\ram,\eta}\).  The definition of \(\loss\) gives its nonnegativity and vanishing, and Problems~\ref{pr:harmonic} and~\ref{pr:final} bound its weighted sum.  This gives the theorem of this chapter.

\begin{theorem}\label{thm:divisibility}
There are real numbers \(\err(p)\), \(p \in \mathbb N\), with the following properties.
\begin{enumerate}
\item[(a)] \(\err(p) \ge 0\) for every \(p\).
\item[(b)] If \(p \ne N\), and \(\chi_K(p) = 1\) or \(x_1 \le p\), then \(\err(p) = 0\).
\item[(c)] \(\sum_{p \in A} \err(p)\log p \le U\) for every finite set \(A\) of natural numbers.
\item[(d)] For every prime \(p\) and every ring map \(\eta\colon \C \to \C_p\) there are an integer \(l \ge 0\) with \(l \le \err(p)\, D_X\) and an \(\Oc_\eta\)-submodule \(S' \subseteq S_{\ram,\eta}\) whose colength in \(S_{\ram,\eta}\) divides \(p^l\), such that for every twist \(\vartheta\), every chain datum \(\Psi\) for \(\vartheta\) with \(p \nmid \Nm Q_\vartheta\), and every \(\varphi \in S'\), the form \(B_{\vartheta,\Psi}\varphi\) has valuation at least \(\gain(p) - \err(p)\).
\end{enumerate}
\end{theorem}

Its proof takes \(\err = \loss\).  The dimension in~(d) is that of all source forms, while the colength is measured in the integral ramified ones; the module \(S'\) depends on \(p\) and \(\eta\) and serves every twist of level prime to \(p\) and every chain datum for that twist.

Theorem~\ref{thm:divisibility} is the divisibility input to Theorem~A: Chapter~\ref{ta-chap} combines~(c) and~(d) with the seesaw identity of Chapter~\ref{ss-chap} and further inputs to bound the index of finite families in the images \(A_s(S_\ram^{\mathrm{int}})\) of the integral ramified source forms under the normalized translates.  Chapters~\ref{d1-chap} and~\ref{d2-chap} prove input~(iii), Chapter~\ref{d5-chap} treats the level prime and Chapter~\ref{d3-chap} the exceptional primes; Chapter~\ref{d6-chap} bounds the size of the composite lift for the trivial twist, against which Theorem~A sets this divisibility.

\section*{Hints}

\subsection*{\S\ref{sec:gain}}
\begin{description}
\item[\ref{pr:qone-split}] Show \(d \equiv q_0/2 - 1 \pmod{q_0}\), so \(d \equiv -1\) modulo \(8\) and modulo each odd prime dividing \(q_0\).  Then quadratic reciprocity.
\item[\ref{pr:shape}] \(I_p\) is the set of odd \(e \ge 1\), and \(c_p = 1\).
\item[\ref{pr:gain-zero}] If \(\chi_K(p) = 1\), \(I_p\) is empty.  If \(p \ge x_1\), compare \(y/p^e\) with \(a/x_1\).
\item[\ref{pr:part-at-p}] Compute \(1 - \chi_K(p^e)\) in the three cases inert, ramified, split.  For \(p^e > a\) the term of \(\sfp\) vanishes.
\end{description}

\subsection*{\S\ref{sec:collisions}}
\begin{description}
\item[\ref{pr:collision-odd}] \(\tau\) and \([-p^e]\tau\) have class \(1\) modulo \(q_0\).  For \(p = d\), compute \(-d^e\) modulo \(q_0\) separately for even and odd \(e\), using \(4 \mid q_0\) and the congruence for \(d\) in the hint to Problem~\ref{pr:qone-split}.
\item[\ref{pr:o-example}] Decompose \((\Z/18480)^\times\) by the Chinese remainder theorem and find its elements of odd order.  Use a prime search for the two minima.
\item[\ref{pr:geometric}] At each of the \(f\) places, a geometric series of ratio at most \(1/2\).
\item[\ref{pr:non-exceptional}] Two cases, according as \(p \equiv \pm 1 \pmod m\) (Problem~\ref{pr:geometric} with \(o = 1\)) or not; in the second, Problem~\ref{pr:collision-odd} and \(m \le p^{r_p}\).
\item[\ref{pr:why-mark}] Every \((\tau, 1)\) with \(\tau \in \Phi_a\) is a collision.
\end{description}

\subsection*{\S\ref{sec:chain}}
\begin{description}
\item[\ref{pr:gauge-chain}] The sup norm is ultrametric.
\item[\ref{pr:collision-chain}] Theorem~\ref{thm:plain}; then (ii) and (iii) with \(\Phi' = \varnothing\).
\end{description}

\subsection*{\S\ref{sec:exceptional}}
\begin{description}
\item[\ref{pr:full-mark}] The relation \(v_{[-p^e]\sigma} = \bar v_\sigma\).
\item[\ref{pr:crude}] Problem~\ref{pr:geometric} with \(o = 1\).
\item[\ref{pr:two-pieces}] The ultrametric inequality.
\item[\ref{pr:colength-add}] A product of two matrices.
\item[\ref{pr:budget}] Each \(\tau \in \Phi_a\) induces one prime.  Bound \(j_p\) by \(2mfa/p + 1\).
\end{description}

\subsection*{\S\ref{sec:sum}}
\begin{description}
\item[\ref{pr:pm-one}] \(mc \le p + 1 \le 2p\).
\item[\ref{pr:term}] Problem~\ref{pr:pm-one}, or \(c = p\).
\item[\ref{pr:harmonic}] Each \(c \le x_1\) serves at most the four numbers \(N\), \(mc - 1\), \(c\), \(mc + 1\); give \(N\) the value \(c = 1\).
\item[\ref{pr:final}] Compare \(16\,t(1 + t)\) with \(t^6\).
\end{description}
