\paragraph{Prerequisites.}
Modules over a discrete valuation ring: a submodule of finite index in \(\Zp^n\) is free of rank \(n\), and a matrix over \(\Q\) can be brought to diagonal form by invertible matrices over \(\Zp\) \cite[Ch.~III, \S7]{La}.  From Chapter~\ref{sd-chap}, the principle that size is compared with divisibility.  From Chapter~\ref{da-chap}: the data (scales \(x < x_1 < b\), \(a = xb\), \(L = \log b\); the CM field \(E\) with its set \(\Phi\) of \(f\) infinite places; \(m\), \(k\), the weights \(\kappa\), the principal level \(M = p_0p_1\), with \(p_0 \ne p_1\) primes, and the prime level \(N\)) and the global prime sums and jet sizes recalled in Section~\ref{sec:exponent}.  From Chapter~\ref{bp-chap}: the source forms, the finite group \(\Gb\), the idempotent \(e_0\), the target \(T\) and the Poincar\'e map.  From Chapter~\ref{ss-chap}: the composite lift and the seesaw identity.  From Chapter~\ref{dc-chap}: the divisibility of the composite lift.  Chapter~\ref{tn-chap} treats small-rank examples of single definite lifts and their norms.

Theorem~A says that the ramified source is spanned by integral forms whose translated images have short lifts at the trivial twist.  It also bounds the index of every finite family of translated images of integral ramified forms in terms of the number of translates used.  The last chapter of this part uses both bounds.  The normalization multiplies by a positive rational number \(c_B\), with exponent \(-\lfloor\gain(p)-\err(p)\rfloor\) at each prime \(p\) in the product, including the level prime \(N\); the price of the shortness is paid in index.  We first do this with rational vectors, and then follow the exponents in the rewritten Lean proof.

\section{Index and colength over \texorpdfstring{\(\Z\)}{Z}}\label{sec:index}

Let \(\Zp\) be the ring of rationals with denominator prime to \(p\), and \(v_p\) the \(p\)-adic valuation, normalized by \(v_p(p) = 1\), with \(v_p(0) = +\infty\).  Let \(y = (y_i)_{i \in \mathcal{J}}\) be a finite family in \(\Q^n\).  A square matrix \(u = (u_{ik})_{i,k \in \mathcal{J}}\) over \(\Zp\) \emph{carries \(y\) with loss \(\delta_p\)} if every \(\sum_k u_{ik}y_k\) lies in \(\Zp^n\) and \(v_p(\det u) \le \delta_p\).  The family \emph{has index at most \(\delta\)} if there are real numbers \(\delta_p \ge 0\) with \(\sum_p \delta_p\log p \le \delta\) such that for every prime \(p\) some matrix carries \(y\) with loss \(\delta_p\).

\begin{problem}\label{pr:one-sixth}
The family \((1/6)\) in \(\Q\) has index at most \(\log 6\), and not at most \(\delta\) for any \(\delta < \log 6\).
\end{problem}

\begin{problem}\label{pr:index-lattice}
A finite family \(y\) in \(\Q^n\) has index at most \(\delta\) if and only if \(\delta \ge \log\,[\Lambda_y : \Z^n]\), where \(\Lambda_y = \Z^n + \sum_i \Z y_i\).  For the family \(((\tfrac12, \tfrac12), (0, \tfrac14))\) in \(\Q^2\) one has \([\Lambda_y : \Z^2] = 8\).
\end{problem}

\begin{problem}\label{pr:indices-add}
If \(\mathcal{J} = \mathcal{J}' \sqcup \mathcal{J}''\) and the two subfamilies have index at most \(\delta'\) and \(\delta''\), then \(y\) has index at most \(\delta' + \delta''\).  Hence a family that is the union of \(t\) subfamilies of index at most \(\delta\) has index at most \(t\delta\).  For the families \((\tfrac12, \tfrac13)\) and \((\tfrac12, \tfrac12)\) in \(\Q\), split into singletons, the bound is \(\log 6\) and \(2\log 2\), and the index is \(\log 6\) and \(\log 2\).
\end{problem}

\begin{problem}\label{pr:colength}
Let \(W\) be a submodule of \(\Zp^k\) with \(\Zp^k/W\) of order \(p^j\).  For every finite family \((\varphi_i)_{i \in \mathcal{J}}\) in \(\Zp^k\) there is a square matrix \(u\) over \(\Zp\) with \(v_p(\det u) \le j\) and \(\sum_k u_{ik}\varphi_k \in W\) for every \(i\).
\end{problem}

\begin{problem}\label{pr:colength-index}
Let \(A\colon \Q^k \to \Q^n\) be linear.  Suppose that for every prime \(p\) there is a submodule \(W_p \subseteq \Zp^k\) of index \(p^{j_p}\) with \(A(W_p) \subseteq \Zp^n\), and \(j_p = 0\) for almost all \(p\).  Then every finite family in \(A(\Z^k)\) has index at most \(\sum_p j_p\log p\).
\end{problem}

\section{A model of Theorem A}\label{sec:model}

Let \(B\colon \Q^k \to \Q^n\) be linear; it stands for the chain of lifts.  Let \(\Pi\) be a finite set of primes, and suppose that for each \(p \in \Pi\) we are given a real number \(\gamma_p\) and a submodule \(W_p \subseteq \Zp^k\) of index \(p^{j_p}\) on which \(B\) is divisible: \(v_p(B\varphi) \ge \gamma_p\) coordinatewise for \(\varphi \in W_p\).  For \(p \notin \Pi\) put \(\gamma_p = j_p = 0\) and \(W_p = \Zp^k\), and suppose \(B(\Zp^k) \subseteq \Zp^n\) for these \(p\).  Put
\[
  g_p = \lfloor \gamma_p \rfloor, \qquad c_B = \prod_{p \in \Pi} p^{-g_p}, \qquad A = c_B B .
\]

\begin{problem}\label{pr:cB-local}
For every prime \(p\), \(v_p(c_B) = -g_p\) and \(A(W_p) \subseteq \Zp^n\).  Hence every finite family in \(A(\Z^k)\) has index at most \(\sum_p j_p \log p\), and \(\lvert A\varphi\rvert^2 = c_B^2\lvert B\varphi\rvert^2\) with \(2\log c_B \le -2\sum_{p \in \Pi} \gamma_p\log p + 2\sum_{p \in \Pi}\log p\).
\end{problem}

\begin{problem}\label{pr:model}
Let \(k = n = 2\), \(\Pi = \{2, 3\}\) and \(B(\varphi_1, \varphi_2) = (4\varphi_1 + 6\varphi_2, 12\varphi_2)\).
\begin{enumerate}
\item With \(\gamma_2 = 1\), \(W_2 = \Z_{(2)}^2\) and \(\gamma_3 = 1\), \(W_3 = \{\varphi : 3 \mid \varphi_1\}\), one has \(c_B = 1/6\); the least \(\delta\) such that \((Ae_1, Ae_2)\) has index at most \(\delta\) is \(\log 3\), and \(\lvert Ae_1\rvert^2 = 4/9\), \(\lvert Ae_2\rvert^2 = 5\).
\item With \(\gamma_2 = 2\), \(W_2 = \{\varphi : 2 \mid \varphi_2\}\) and the same choice at \(3\), one has \(c_B = 1/12\); this least \(\delta\) is \(\log 6\), and \(\lvert Ae_1\rvert^2 = 1/9\), \(\lvert Ae_2\rvert^2 = 5/4\).
\end{enumerate}
\end{problem}

In Problem~\ref{pr:model}, increasing \(\gamma_2\) by one halves \(c_B\), shortens the image, and increases the index of the family by \(\log 2\).  For each prime \(p\) and each ring homomorphism \(\C \to \C_p\), the divisibility theorem of Chapter~\ref{dc-chap} gives a submodule of the integral ramified source forms of colength exponent at most \(\err(p)\) times the dimension of the full source.  On this submodule the composite lift has valuation at least \(\gain(p) - \err(p)\) for every twist of level prime to \(p\) and every chain datum for that twist.

\section{The exponent}\label{sec:exponent}

We recall the global prime sums and sizes from Chapter~\ref{da-chap} and the local gains from Chapter~\ref{dc-chap}, for the data of the rewritten Lean proof \cite{GL}.  The places \(\Phi\) are identified with \((\Z/q_1)^\times\), and for \(\nu \in \Z\) let \(\langle\nu\rangle \in \Phi\) be the place of the class of \(\nu\) if \(\nu\) is prime to \(q_1\), and the identity otherwise.  The set \(\Phi_a\) consists of the places \(\equiv 1 \pmod{q_0}\).  For real \(y\), \(\sigma \in \Phi\) and \(j \in \mathbb{N}\), with \(z_+ = \max(z, 0)\) and \(\chi_K(n) = (n/d)\),
\[
  \sfin(y, \sigma) = \sum_{\nu = 1}^{a} \Lambda(\nu)\bigl(1 - \chi_K(\nu)\bigr)\Bigl(\frac{y}{\nu} - \kappa(\langle -\nu\rangle\sigma)\Bigr)_+ , \qquad \sinf(j) = \log j! + j\log\frac{M}{\Omega} .
\]
Here \(\Omega = e^{\mathrm{CS}(d)}\,2\pi/\mathrm{rd}_E\), with \(\mathrm{rd}_E\) the root discriminant of \(E\), and the constant \(c_1\) of the prime sum satisfies \(c_1 = -1 - \mathrm{CS}(d)\).  Put \(w = f^{-1}\sum_{\tau \in \Phi}\log\kappa(\tau)\), \(\lambda_E = \log(M\,\mathrm{rd}_E/2\pi)\), \(s_0 = a(\log a - w + c_1)\), and
\[
  \hm = mk\sum_{\tau \in \Phi}\bigl(\sinf(\kappa(\tau)) - \sfin(\kappa(\tau), \tau)\bigr), \qquad
  \rem = \sum_{\tau \in \Phi_a}\bigl(\sfin(a, \tau) - s_0\bigr).
\]
For a prime \(p\) let \(c_p = \frac12\) if \(p = d\) and \(c_p = 1\) otherwise, and let \(\mathcal{E}_p\) be the set of integers \(e \ge 1\) such that \(p = d\), or \(\chi_K(p) = -1\) and \(e\) is odd.  Put
\[
  s_p(y, \sigma) = c_p\sum_{e \in \mathcal{E}_p}\Bigl(\frac{y}{p^e} - \kappa(\langle -p^e\rangle\sigma)\Bigr)_+ , \qquad
  \gain(p) = m\Bigl(\sum_{\tau \in \Phi_a}s_p(a, \tau) + k\sum_{\sigma \in \Phi}s_p(\kappa(\sigma), \sigma)\Bigr).
\]
Let \(\Pc\) be the set of primes \(p < x_1\) with \(\chi_K(p) \ne 1\), and \(\Sigma = \sum_{p \in \Pc}\gain(p)\log p\).  The unit of loss is \(U = faL^3\).  We use the following properties of the data: \(x_1 \le a\); \(a/x_1 \le \kappa(\tau) \le a\) for every \(\tau\); \(\sinf(j) \ge 0\) for \(j \le a\); \(\log N \ge mw + (m - 1)\lambda_E - \frac{15}{2}L\); \(m(1 + \frac12\log a) \le aL\); and \(\lambda_E \le 23L\).

\begin{problem}\label{pr:weight-prime-power}
For a prime power \(\nu = p^e\), \(e \ge 1\), one has \(\Lambda(\nu)(1 - \chi_K(\nu)) = 2c_p\log p\) if \(e \in \mathcal{E}_p\), and \(0\) otherwise.
\end{problem}

\begin{problem}\label{pr:prime-sum-local}
For \(y \le a\) and \(\sigma \in \Phi\), \(\sfin(y, \sigma) = \sum_{p < x_1} 2\,s_p(y, \sigma)\log p\), the sum over primes.
\end{problem}

\begin{problem}\label{pr:gain-zero}
If \(\chi_K(p) = 1\) or \(p \ge x_1\), then \(\gain(p) = 0\).  Consequently
\[
  2\Sigma = m\sum_{\tau \in \Phi_a}\sfin(a, \tau) + mk\sum_{\tau \in \Phi}\sfin(\kappa(\tau), \tau) .
\]
\end{problem}

\begin{problem}\label{pr:exponent-identity}
\(\hm - m\,\rem + 2\Sigma = m\,\#\Phi_a\, s_0 + mk\sum_{\tau \in \Phi}\sinf(\kappa(\tau))\).
\end{problem}

The prime sums have cancelled.  The long directions contribute \(m\,\#\Phi_a\,s_0\) to the right-hand side.

\begin{problem}\label{pr:stirling}
For every integer \(n \ge 1\), \(\log n! \le n\log n - n + 1 + \frac12\log n\).
\end{problem}

\begin{problem}\label{pr:excess}
\(\log(M/\Omega) = 1 + c_1 + \lambda_E\), and \(\sinf(a) - s_0 \le a(w + \lambda_E) + 1 + \frac12\log a\).
\end{problem}

\begin{problem}\label{pr:level-pays}
\(m(\sinf(a) - s_0) - a\log N \le 32\,aL\).
\end{problem}

At each long place, the level term \(a\log N\) covers the excess of the \(m\) lifts up to the allowance \(32aL\).  Put \(C_{\mathrm{ch}} = m\bigl(\#\Phi_a\,\sinf(a) + k\sum_{\tau \in \Phi}\sinf(\kappa(\tau))\bigr)\).

\begin{problem}\label{pr:chain-size}
\(C_{\mathrm{ch}} - \#\Phi_a\,a\log N \le \hm - m\,\rem + 2\Sigma + 32\,faL\).
\end{problem}

The composite lift \(B_{\vartheta,\Psi}\) of Chapter~\ref{ss-chap} is defined on all source forms; for a twist \(\vartheta\) it depends on a chain datum \(\Psi\) for \(\vartheta\), a finitely supported function on the translations.  For a form \(F\) on \(X_0\) or on \(Y_m\) and \(\sigma \in \Aut(\C)\), \(\lvert\sigma F(g)\rvert^2\) denotes the Weyl squared norm of the conjugate value.  The rewritten Lean proof \cite{GL} proves the following.

\begin{theorem}\label{thm:size}
Let \(\Psi\) be a chain datum for the trivial twist, \(\sigma \in \Aut(\C)\), \(\varphi\) a source form and \(\xi \in \R\).  If \(\lvert\sigma\varphi(g)\rvert^2 \le e^\xi\) for every \(g \in G_f(X_0)\), then \(\lvert\sigma(B_{1,\Psi}\varphi)(y)\rvert^2 \le \exp(\xi + \beta)\) for every \(y \in G_f(Y_m)\), where
\[
  \beta = \hm - m\,\rem + 2\Sigma + U .
\]
\end{theorem}

After expanding the chain datum, the rewriting bounds the product of the \(m\) kernels.  If \(R_j\) is the sum of \(\sinf\) over the row lengths at step \(j\), the contribution to the squared-norm exponent is at most \(2mx^{50} + \sum_{j<m}R_j\), with \(\sum_{j<m}R_j \le C_{\mathrm{ch}} + U/2\).  The restriction to \(Y_m\) contributes a saving of \(\#\Phi_a\,a\log N\) up to an error.  The other half of \(U\) covers the kernel, class-count, coefficient and restriction errors and the term \(32faL\) of Problem~\ref{pr:chain-size}.  The theorem is a statement at the trivial twist only.

\section{Theorem A}\label{sec:theorem-a}

We recall the objects of Chapters~\ref{bp-chap} and~\ref{ss-chap}.  The source forms make up a space \(S_{\mathrm{all}}\) of dimension \(D_X\), and \(S \subseteq S_{\mathrm{all}}\) is the space of ramified source forms.  For a pair \(\iota\) of a prime \(p\) and an isomorphism \(\C \simeq \bar\Q_p\), let \(\Oc_\iota\) be the ring of \(z \in \C\) with \(\lvert\iota(z)\rvert_p \le 1\).  A form is integral at a ring homomorphism \(\eta\colon \C \to \C_p\) if all its values have gauge at most \(1\) at \(\eta\), and integral at \(\iota\) if it is integral at the homomorphism \(\C \to \C_p\) induced by \(\iota\).  Let \(I\) be the set of \(\varphi \in S\) integral at every \(\iota\).  A form \(\varphi\) is \(e^U\)-short if it is integral at every \(\eta\colon \C \to \C_p\), for every prime \(p\), and \(\lvert\sigma\varphi(g)\rvert^2 \le e^U\) for all \(\sigma \in \Aut(\C)\) and all \(g\).  Let \(\rho\) be the action of \(\Gb\) on the principal-level forms on \(Y_0\), \(T\) the image of \(\rho(e_0)\) in that space, and \(\mathfrak{P}\) the Poincar\'e map from \(S_{\mathrm{all}}\) to the principal-level forms.  A tame character has the data of a twist as in Chapter~\ref{ss-chap}, with the tame part permitted to be trivial also at places of its set \(Q\); its level is the product of the residue-field sizes at the places of \(Q\).  For every tame character \(\mu\) let \(\Xi_\mu\) be the lower chain of \(m\) lifts from \(Y_0\) to \(Y_m\); when taking its class norm or testing integrality we view its value at the prescribed last level and character.  For a twist \(\vartheta\) write \(\Xi_\vartheta\) for the chain at its underlying tame character.  An \(m\)-fold twist is a twist whose tame part \(\psi_w\) satisfies \(\psi_w^i \ne 1\) for every \(w \in Q\) and \(1 \le i \le m\) (Chapter~\ref{tn-chap}, Section~\ref{tn-sec:chain}).  For \(y \in T\) and an \(m\)-fold twist \(\vartheta\), \(\Theta_\vartheta y\) is \(\Xi_\vartheta y\) at the level of \(\vartheta\), and \(\lVert F\rVert_\sigma^2\) is the sum over classes of the squared norms \(\lvert\sigma F(g)\rvert^2\), each class weighted by the inverse order of its stabilizer.  For \(c \in \C\) the squared norm of \(\sigma(cF(g))\) is \(\lvert\sigma(c)\rvert^2\) times that of \(\sigma F(g)\), and the gauge of \(cF(g)\) at \(\eta\) is \(\lvert\eta(c)\rvert_p\) times that of \(F(g)\).

The text uses the following results of the rewritten Lean proof \cite{GL}.
\begin{enumerate}
\item The \(e^U\)-short forms in \(S\) span \(S\).
\item \emph{Seesaw.}  There is \(c_{\mathrm{ss}} \in \C^\times\) such that for every twist \(\vartheta\) there is a chain datum \(\Psi\) for \(\vartheta\) with \(\Xi_\vartheta(\mathfrak{P}\varphi) = c_{\mathrm{ss}}B_{\vartheta,\Psi}\varphi\) for all \(\varphi \in S\) (Chapter~\ref{ss-chap}).
\item \emph{Good part.}  Let \(z\) be a form of principal level on \(Y_0\) and \(s \in \Gb\).  If \(\lvert\sigma(\Xi_1 z)(g)\rvert^2 \le e^\xi\) for all \(g \in G_f(Y_m)\), with \(\xi \in \R\) and a fixed \(\sigma \in \Aut(\C)\), then \(\lVert\Xi_1(\#\Gb\,\rho(e_0)\rho(s)z)\rVert_\sigma^2 \le e^{x^{99} + \xi}\).  For every prime \(p\), every ring homomorphism \(\eta\colon \C \to \C_p\), every tame character \(\mu\) and every \(c \in \C\), if \(c\,\Xi_\mu z\) is integral at \(\eta\), so is \(c\,\Xi_\mu(\#\Gb\,\rho(e_0)\rho(s)z)\).
\item \emph{Divisibility} (Chapter~\ref{dc-chap}).  There is a function \(\err\colon \mathbb{N} \to \R_{\ge 0}\) with \(\err(n) = 0\) whenever \(n \ne N\) and either \(\chi_K(n) = 1\) or \(n \ge x_1\), and with \(\sum_{n \in F}\err(n)\log n \le U\) for every finite \(F \subseteq \mathbb{N}\) (the term at \(0\) taken to be zero).  For every prime \(p\) and every ring homomorphism \(\eta\colon \C \to \C_p\), let \(\Oc_\eta\) be the ring of \(z \in \C\) with \(\lvert\eta(z)\rvert_p \le 1\) and \(M_\eta\) the \(\Oc_\eta\)-module of forms in \(S\) integral at \(\eta\).  There are an integer \(j_{p,\eta} \ge 0\) with \(j_{p,\eta} \le \err(p)D_X\) and an \(\Oc_\eta\)-submodule \(S'_{p,\eta} \subseteq M_\eta\) such that every finite family in \(M_\eta\) is carried into \(S'_{p,\eta}\) by a square matrix over \(\Oc_\eta\) whose determinant divides \(p^{j_{p,\eta}}\), and such that for every twist \(\vartheta\) of level prime to \(p\), every chain datum \(\Psi\) for \(\vartheta\) and every \(\varphi \in S'_{p,\eta}\), the gauge of \((B_{\vartheta,\Psi}\varphi)(g)\) at \(\eta\) is at most \(p^{\err(p) - \gain(p)}\) for every \(g \in G_f(Y_m)\).
\item \(\dim S \ge (1 - f/(N - 1))D_X\), and \(f/(N - 1) \le b^{-8}\).
\end{enumerate}
Moreover the data satisfy \(b \ge 2\), \(N \ge x_1\), \(\sum_{p \in \Pc \cup \{N\}}\log p \le b\) and \(x^{99} + 4U + 2b \le faL^4\).

Fix the witness \(\err\) of input~4 chosen in the rewriting.  As in Section~\ref{sec:model}, put \(g_p = \lfloor\gain(p) - \err(p)\rfloor\) and \(c_B = \prod_{p \in \Pc \cup \{N\}}p^{-g_p}\), a positive rational number, and for \(s \in \Gb\) define \(A_s\colon S \to T\) by
\[
  A_s\varphi = c_B\,c_{\mathrm{ss}}^{-1}\,\#\Gb\,\rho(e_0)\rho(s)\,\mathfrak{P}\varphi .
\]

\begin{problem}\label{pr:cB-general}
\(\log c_B \le -\Sigma + U + b\).
\end{problem}

\begin{problem}\label{pr:cB-padic}
For every prime \(p\) and every \(\iota\) above \(p\), \(\lvert\iota(c_B)\rvert_p = p^{g_p}\), and \(g_p = 0\) for \(p \notin \Pc \cup \{N\}\).
\end{problem}

\begin{problem}\label{pr:exponent-a}
\((U + \beta) + x^{99} + 2\log c_B \le \hm - m\,\rem + x^{99} + 4U + 2b\).
\end{problem}

\begin{problem}\label{pr:short-lifts}
Every \(e^U\)-short \(\varphi \in S\) lies in \(I\) and satisfies \(\lVert\Theta_1 A_s\varphi\rVert_\sigma^2 \le \exp(\hm - m\,\rem + faL^4)\) for all \(s \in \Gb\) and \(\sigma \in \Aut(\C)\).
\end{problem}

\begin{problem}\label{pr:dim-ramified}
\(D_X \le 2\dim S\).
\end{problem}

In the twisted target lattice of the rewritten Lean proof \cite{GL}, a vector \(y \in T\) is integral at \(\iota\) above \(p\) if \(\Theta_\vartheta y\) is integral at \(\iota\) for every \(m\)-fold twist \(\vartheta\) of level prime to \(p\).  A finite family \(y\) in \(T\) has index at most \(\delta\) if there are \(\delta_p \ge 0\) with \(\sum_{p \in F}\delta_p\log p \le \delta\) for every finite set \(F\) of primes, such that for every \(\iota\) above every \(p\) some square matrix \(u\) over \(\Oc_\iota\) with \(\lvert\iota(\det u)\rvert_p \ge p^{-\delta_p}\) carries \(y\) into the vectors integral at \(\iota\).  This is the carrying condition of Section~\ref{sec:index} at each \(\iota\), with one matrix carrying the family into the vectors integral for all allowed twists at once.

\begin{problem}\label{pr:integral-image}
Let \(\iota\) lie above \(p\), and let \(\eta\) be the homomorphism induced by \(\iota\).  Then \(A_s\varphi\) is integral at \(\iota\) for every \(\varphi \in S'_{p,\eta}\) and every \(s \in \Gb\).
\end{problem}

\begin{problem}\label{pr:index-one-translate}
For each \(s \in \Gb\), every finite family in \(A_s(I)\) has index at most \(\dim S \cdot 2U\).
\end{problem}

The rewritten Lean proof \cite{GL} proves the following two statements.

\begin{theorem}[Theorem A]\label{thm:A}
\begin{enumerate}
\item[(a)] The complex span of the set of \(\varphi \in I\) such that
\[
  \lVert\Theta_1 A_s\varphi\rVert_\sigma^2 \le \exp\bigl(\hm - m\,\rem + faL^4\bigr) \quad\text{for all } s \in \Gb \text{ and } \sigma \in \Aut(\C)
\]
is \(S\).
\item[(b)] Let \(t \subseteq \Gb\) be finite, \(n \in \mathbb{N}\), and \(y_1, \dots, y_n \in T\).  If each \(y_i\) lies in \(A_s(I)\) for some \(s \in t\), then \((y_1, \dots, y_n)\) has index at most \(\#t\cdot\dim S\cdot 2faL^3\).
\end{enumerate}
\end{theorem}

Neither part asks the forms to be independent or the images to span; Theorem~B of Chapter~\ref{bp-chap} supplies few translates whose images span almost all of \(T\).  In the exponent of~(a) the gains have disappeared and the remainder appears with the factor \(m\).  The last chapter of this part compares this upper bound with a lower bound in which the main amount \(\hm\) appears without the remainder.

\paragraph{In the Lean formalization.}

\section*{Hints}

\subsection*{\S\ref{sec:index}}
\begin{description}
\item[\ref{pr:one-sixth}] A \(1 \times 1\) matrix \(u\) carries \(1/6\) at \(p\) only if \(v_p(u) \ge -v_p(1/6)\).
\item[\ref{pr:index-lattice}] Write the matrix of the family as \(PDQ\) with \(P, Q\) invertible over \(\Zp\) and \(D\) diagonal.
\item[\ref{pr:indices-add}] Block diagonal matrices; Problem~\ref{pr:index-lattice} for the examples.
\item[\ref{pr:colength}] Pull \(W\) back along \(\Zp^{\mathcal{J}} \to \Zp^k\), \(e_i \mapsto \varphi_i\), and take a basis of the preimage.
\item[\ref{pr:colength-index}] Choose preimages of the family in \(\Z^k\).  Which local matrices can act on these preimages?
\end{description}

\subsection*{\S\ref{sec:model}}
\begin{description}
\item[\ref{pr:cB-local}] How does multiplication by \(c_B\) change each coordinate valuation on \(W_p\)?
\item[\ref{pr:model}] Problem~\ref{pr:index-lattice}.
\end{description}

\subsection*{\S\ref{sec:exponent}}
\begin{description}
\item[\ref{pr:prime-sum-local}] Problem~\ref{pr:weight-prime-power}; the terms with \(p^e \ge x_1\) vanish because \(\kappa \ge a/x_1\).
\item[\ref{pr:gain-zero}] Consider \(\mathcal{E}_p\) in the split case and the positive part in the large-prime case.  Which two choices of \(y\) give the sums in \(\gain(p)\)?
\item[\ref{pr:exponent-identity}] Problem~\ref{pr:gain-zero}.
\item[\ref{pr:stirling}] Show that \(\log n! - (n + \frac12)\log n + n\) decreases in \(n\); the series of \(\frac12\log\frac{1+t}{1-t}\) helps.
\item[\ref{pr:excess}] Problem~\ref{pr:stirling} with \(n = a\).
\item[\ref{pr:level-pays}] Compare Problem~\ref{pr:excess} with the lower bound for \(\log N\).  Which terms of size \(ma(w + \lambda_E)\) cancel?
\item[\ref{pr:chain-size}] Problems~\ref{pr:exponent-identity} and~\ref{pr:level-pays}, and \(\#\Phi_a \le f\).
\end{description}

\subsection*{\S\ref{sec:theorem-a}}
\begin{description}
\item[\ref{pr:cB-general}] \(-g_p \le -\gain(p) + \err(p) + 1\), and \(\gain(N) = 0\).
\item[\ref{pr:cB-padic}] Which factors of \(c_B\) can have nonzero valuation at \(p\)?  Compare with the support conditions on the gain and error.
\item[\ref{pr:exponent-a}] Problem~\ref{pr:cB-general}.
\item[\ref{pr:short-lifts}] The seesaw at the trivial twist, Theorem~\ref{thm:size} with \(\xi = U\), the good part, and Problem~\ref{pr:exponent-a}.
\item[\ref{pr:dim-ramified}] Input~5.
\item[\ref{pr:integral-image}] Fix an \(m\)-fold twist of level prime to \(p\).  Compare the gauge bound from the seesaw and divisibility with the local absolute value of \(c_B\), then use the good-part input.
\item[\ref{pr:index-one-translate}] As in Problem~\ref{pr:colength-index}, with \(\delta_p = D_X\err(p)\); then Problem~\ref{pr:dim-ramified}.
\end{description}
