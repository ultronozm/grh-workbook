\title{Free levels and short bases}

\paragraph{Prerequisites.}
Hermitian lattices over the integers of a CM field, finite adeles, the different and the trace form of a number field.  Lattices in \(\R^m\) and their duals.  Weyl modules of \(\mathrm{GL}_n\) at the level of highest weights.  From Chapter~\ref{ob-chap}: the standard lattice stabilizer \(\Ko(V)\) (Section~\ref{ob-sec:spaces}), the principal level (Section~\ref{ob-sec:forms}), the gauge and integrality at a ring homomorphism \(\eta\colon\C\to\C_p\) (Definition~\ref{ob-def:gauge}), the source forms \(S\) (Section~\ref{ob-sec:source}) and the target lattice (Definition~\ref{ob-def:target-lattice}).  From Chapter~\ref{ta-chap}: \(Z\)-short forms (Section~\ref{ta-sec:theorem-a}).  From Chapter~\ref{dt-chap}: the Hermitian \(\Z[i]\)-lattice \(\Z[i]^n\) is alone in its genus for \(n\le4\) (Problem~\ref{dt-pr:class-one}).  From Chapter~\ref{da-chap}: the data.  Pairing conditions of twisted lattices: Chapter~\ref{at-chap}, Section~\ref{at-sec:toy}.

\medskip
Input~1 of Theorem~\ref{ta-thm:A} (Chapter~\ref{ta-chap}) rests on one statement: the \(4^{-f}e^U\)-short source forms span \(S\), with \(U\) as in Section~\ref{da-sec:amounts}.  Why should short forms span at all?  If the group \(\Gamma\) of rational unitary matrices acts freely on the cosets \(gK\) of the level \(K\), a form is determined by its values at representatives of the double classes, and those values may be prescribed arbitrarily.  It is then enough to find, at each point \(g\), a basis of the Weyl module consisting of vectors that are integral at \(g\) and short at every conjugate.  Section~\ref{sec:free} treats such levels, Section~\ref{sec:short} uses transference to find short \(E\)-bases of \(E^n\) in the lattice of \(g\), and Section~\ref{sec:eval} turns them into short vectors of a Weyl module.

\section{Free levels}\label{sec:free}

Let \(E\) be a CM field with a CM type, \(f\) its number of infinite places \(\tau\), \(\Phi_\tau\) the chosen embeddings, \(z\mapsto\bar z\) complex conjugation, \(\Oh\) the integral finite adeles.  For \(n\ge1\), \(G_f\) is the finite adelic unitary group of the standard definite form \(\sum_i\bar x_iy_i\) on \(E^n\), \(\Gamma\) its group of rational points, and \(\Ko = G_f\cap\mathrm{GL}_n(\Oh)\) the stabilizer of \(\Oh^n\), as in Section~\ref{ob-sec:spaces}.  A \emph{level} is a subgroup \(K\subseteq\Ko\) containing a neighbourhood of the identity; its \emph{classes} are the finitely many double cosets \(\Gamma gK\).  For a nonzero ideal \(\mathfrak a\) of \(\Oc_E\), \(\Ko(\mathfrak a)\) consists of the elements of \(\Ko\) that reduce to \(1\) modulo \(\mathfrak a\).

A \emph{weight} \(\kappa=(\kappa_\tau)\) is a dominant weight \(\kappa_{\tau,1}\ge\dots\ge\kappa_{\tau,n}\) of \(\mathrm{GL}_n\) at each \(\tau\); put \(\lvert\kappa\rvert_{<n}:=\sum_\tau\sum_i(\kappa_{\tau,i}-\kappa_{\tau,n})\).  Its Weyl module \(W_\kappa\) (Section~\ref{sec:eval}) carries the action \(\rho_\kappa\) of \(\prod_\tau\mathrm{GL}_n\), the gauge of Definition~\ref{ob-def:gauge}, and the Weyl squared norm \(\lvert\cdot\rvert^2\), invariant under \(\prod_\tau U(n)\).  A form of weight \(\kappa\) and level \(K\) is a function \(F\colon G_f\to W_\kappa\) with \(F(\gamma gk)=\gamma F(g)\) for \(\gamma\in\Gamma\), \(k\in K\), where \(\gamma\) acts by \(\rho_\kappa((\Phi_\tau(\bar\gamma))_\tau)\), with conjugation and each embedding applied entrywise, as for the source forms (Section~\ref{ob-sec:source}).

\begin{definition}\label{def:free}
A level \(K\) is \emph{free} if \(\Gamma\cap gKg^{-1}=\{1\}\) for every \(g\in G_f\).
\end{definition}

\begin{problem}\label{pr:units}
\(U(2)(\Z[i])\) consists of the \(32\) monomial matrices whose nonzero entries belong to \(\{\pm1,\pm i\}\); \(16\) of them are \(\equiv1\bmod(1+i)\), and only \(1\) is \(\equiv1\bmod(2+i)\).  Hence on \(\Z[i]^2\) the level \(\Ko(1+i)\) is not free and \(\Ko(2+i)\) is free.
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:units

\begin{problem}\label{pr:minkowski}
Let \(\mathfrak a=w_1\cdots w_s\) be a product of distinct finite places with \(N\mathfrak a>(2n)^{[E:\Q]}\).  If \(\gamma\in\Gamma\), \(g\in G_f\) and \(g^{-1}\gamma g\in\Ko(\mathfrak a)\), then \(\tr\gamma-n\in\mathfrak a\), every conjugate of \(\tr\gamma-n\) has absolute value at most \(2n\), and \(\gamma=1\).  For \(\Q(i)\), \(n=2\), \(\mathfrak a=(2+i)\) one has \(N\mathfrak a=5<16\): the criterion is sufficient only.
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:minkowski

This is a form of Minkowski's lemma on torsion in congruence subgroups \cite{Se}.

\begin{problem}\label{pr:free}
A level contained in \(\Ko(\mathfrak a)\), with \(\mathfrak a\) as in Problem~\ref{pr:minkowski}, is free.  For the data and \(n\le4n_m\), the product of the \(f\) chosen places above \(p_0\) is such an \(\mathfrak a\); hence every level of rank \(n\le4n_m\) whose elements reduce to \(1\) at the chosen places is free, among them the level of the source forms and the levels of the targets of the lower chains.
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:free

\begin{problem}\label{pr:dim-free}
Let \(K\) be free, with class representatives \(g_c\).  Then \(F\mapsto(F(g_c))_c\) is an isomorphism from the forms of weight \(\kappa\) and level \(K\) onto \(\bigoplus_cW_\kappa\); their dimension is \(\#\{\text{classes}\}\cdot\dim W_\kappa>0\).  If \(K'\subseteq K\) are free, \(K'\) has \([K:K']\) times as many classes as \(K\).  On \(\Z[i]^2\), \(\Ko(2+i)\) has \(15\) classes.
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:dim-free

\begin{definition}\label{def:delta}
For \(x\in G_f\) and \(v\in W_\kappa\), define the \emph{delta form} of level \(K\) by \(\varphi_{x,v}(y):=\sum\gamma^{-1}v\), the sum over the \(\gamma\in\Gamma\) with \(x^{-1}\gamma y\in K\).
\end{definition}

\begin{problem}\label{pr:delta}
At a free level the sum defining \(\varphi_{x,v}(y)\) has at most one term; \(\varphi_{x,v}\) is the form with value \(v\) at \(x\) that vanishes off \(\Gamma xK\).  At every \(y\in\Gamma xK\), the gauge of \(\varphi_{x,v}(y)\) at \(y\) equals that of \(v\) at \(x\), for every \(\eta\), and \(\lvert\sigma\varphi_{x,v}(y)\rvert^2=\lvert\sigma v\rvert^2\) for every \(\sigma\in\Aut(\C)\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:delta

\begin{problem}\label{pr:full-level}
At the level \(\Ko\) of \(\Z[i]^2\) (not free), write \(\Sym^d\) as forms of degree \(d\) in \(e_1,e_2\).  Then \(\sum_\gamma\gamma^{-1}e_1^4=16(e_1^4+e_2^4)\) and \(\sum_\gamma\gamma^{-1}e_1^2=0\), the sums over \(U(2)(\Z[i])\).  The forms of level \(\Ko\) and weight \(\Sym^4\) form a line, and those of weight \(\Sym^2\) are \(0\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:full-level

For forms \(F,F'\) of the same weight and level \(K\) put \(\langle F,F'\rangle:=\sum_c\#(\Gamma\cap g_cKg_c^{-1})^{-1}\langle F(g_c),F'(g_c)\rangle_W\), with the Weyl inner product, conjugate-linear in the first argument.  We use one fact read from the proof \cite{GL}: if \(u\) has gauge at most \(1\) at \(g\) for \(\eta\circ(z\mapsto\bar z)\) and \(v\) has gauge at most \(1\) at \(g\) for \(\eta\), then \(\lvert\eta\langle u,v\rangle_W\rvert\le1\).  \emph{Status}: read from the proof; not proved here.

\begin{problem}\label{pr:pairing}
At a free level, \(\langle F,F'\rangle=\sum_c\langle F(g_c),F'(g_c)\rangle_W\).  If \(F\) is integral at \(\bar\iota\) and \(F'\) at \(\iota\), then \(\langle F,F'\rangle\in\Oc_\iota\).  The same conclusions hold for two forms transforming on the right under the same character \(\chi\) of \(K\) with \(\lvert\chi\rvert=1\).  Hence the target lattice satisfies the pairing condition of a twisted lattice (Chapter~\ref{at-chap}, Section~\ref{at-sec:toy}).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:pairing

\section{Short bases}\label{sec:short}

For \(g\in G_f\) let \(\Lambda(g):=E^n\cap g\Oh^n\), and for \(y\in E^n\) let \(\lvert y\rvert^2:=\sum_\tau\sum_i\lvert\Phi_\tau(y_i)\rvert^2\).  Through \(y\mapsto(\Phi_\tau(y_i))_{\tau,i}\), \(\Lambda(g)\) is a lattice of rank \(2fn\) in \(\C^{fn}=\R^{2fn}\), with the inner product \(\operatorname{Re}\sum\bar x_jy_j\).  Fix a positive integer \(N'\) in the different of \(E\).

\begin{problem}\label{pr:first-min}
Every nonzero \(y\in\Lambda(g)\) has \(\lvert y\rvert\ge1\).  For \(E=\Q(i)\), \(n=1\) and \(g\) with components \(5\) and \(5^{-1}\) at \(2+i\) and \(2-i\) and \(1\) elsewhere, \(\Lambda(g)=\frac{2+i}{2-i}\Z[i]\), and the bound is attained.
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:first-min

\begin{problem}\label{pr:dual}
Let \(\mathfrak D_E\) be the different of \(E\).  The dual of \(\Lambda(g)\) for the trace pairing \((a,\xi)\mapsto\tr_{E/\Q}(a^*\xi)\) on \(E^n\) is \(\mathfrak D_E^{-1}\Lambda(g)\).  The Euclidean dual of \((2/N')\Lambda(g)\) is \(N'\mathfrak D_E^{-1}\Lambda(g)\), and no nonzero vector of this dual is shorter than \(1\).  For \(E=\Q(i)\) one may take \(N'=2\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:dual

\emph{Transference.}  Let \(\Lambda\subset\R^m\) be a lattice whose dual has no nonzero vector shorter than \(r_0>0\).  Then every point of \(\R^m\) lies within \(m^2/r_0\) of \(\Lambda\).  Banaszczyk \cite{Ba} proves the bound \(m/(2r_0)\); bounds of this shape go back to \cite{LLS}.  \emph{Status}: classical.

\begin{problem}\label{pr:transference-example}
For \(\Lambda=\Z(1,0)+\Z(0,10)\) the shortest nonzero dual vector has length \(1/10\), and the largest distance of a point from \(\Lambda\) is \(\sqrt{101}/2=5.02\ldots\le40\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:transference-example

\begin{problem}\label{pr:cover-span}
If every point of \(\R^m\) lies within \(r>0\) of a set \(\mathcal S\), the vectors of \(\mathcal S\) of length at most \(3r\) span \(\R^m\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:cover-span

\begin{definition}\label{def:short-basis}
Put \(\csb:=6N'(fn)^2\).  A \emph{short basis} at \(g\) is a matrix \(\B(g)\in\mathrm{GL}_n(E)\) such that \(g^{-1}\B(g)\) has entries in \(\Oh\) and \(\lvert\phi(\B(g)_{ij})\rvert\le\csb\) for all \(i,j\) and every embedding \(\phi\colon E\to\C\).
\end{definition}

\begin{problem}\label{pr:short-basis}
Every \(g\in G_f\) has a short basis.  For \(\Q(i)\), \(n=2\), \(N'=2\): \(\csb=48\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:short-basis

\begin{problem}\label{pr:approx}
For every finite adele \(y\) of \(E\) there is \(a\in E\) with \(y-a\in\Oh\) and \(\lvert\phi(a)\rvert\le(2f)^2N'\) for every embedding \(\phi\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:approx

\begin{problem}\label{pr:same-gram}
Let \(Y,Y'\in\mathrm{GL}_n(E)\) have \(Y'^*Y'=Y^*Y\), and let \(g,g'\in G_f\) be such that \((g'^{-1}Y')(g^{-1}Y)^{-1}\) has entries in \(\Oh\).  Then \(\Gamma g\Ko=\Gamma g'\Ko\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:same-gram

\begin{problem}\label{pr:invariant}
Let \(\B=\B(g)\), \(A=g^{-1}\B\), and let \(\mathsf X\in M_n(E)\) have \(A^{-1}-\mathsf X\) integral and \(\lvert\phi(\mathsf X_{ij})\rvert\le (2f)^2N'\) for every embedding \(\phi\) and all \(i,j\), as in Problem~\ref{pr:approx}.  Put \(G:=\B^*\B\).  The entries of \(G\) and \(G\mathsf X\) are algebraic integers whose conjugates have absolute value at most \(\csb^3\), and the pair \((G,G\mathsf X)\) determines the class \(\Gamma g\Ko\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:invariant

\begin{problem}\label{pr:count-integers}
For \(C\ge0\), at most \((1+2C)^{2f}\) algebraic integers of \(E\) have all conjugates of absolute value at most \(C\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:count-integers

Problems~\ref{pr:short-basis}--\ref{pr:count-integers} prove the following.

\begin{theorem}\label{thm:class-count}
Let \(E\) be a CM field with \(f\) infinite places, \(N'\) a positive integer in its different, and \(\csb=6N'(fn)^2\).  The standard definite space of rank \(n\) has at most \((1+2\csb^3)^{4fn^2}\) classes \(\Gamma\backslash G_f/\Ko\).
\end{theorem}

For \(\Q(i)\), \(n=1\), \(N'=2\) the bound is \(3457^4\), and there is one class.  Chapter~\ref{d6-chap} uses this count for the data.

\section{Evaluation vectors and short forms}\label{sec:eval}

Fix \(\tau\) and write \(\lambda=\kappa_\tau-\kappa_{\tau,n}(1,\dots,1)\), a partition whose columns have height \(h<n\).  For a filling \(T\) of the diagram of \(\lambda\) by \(\{1,\dots,n\}\) and an \(n\times n\) matrix \(y\), let \(b_T(y)\) be the product, over the columns \((i_1,\dots,i_h)\) of \(T\) read from top to bottom, of \(\det(y_{i_a,b})_{1\le a,b\le h}\).  The products \(b_T\) span a space \(\nabla_\lambda\) of polynomials, and those with \(T\) semistandard form a \(\Z\)-basis of the \(\Z\)-span of all the \(b_T\) \cite{DEP}, \cite{Fu}; \emph{Status}: classical.  We take \(W_\kappa:=(\bigotimes_\tau\nabla_{\lambda_\tau})^*\), with coordinates dual to the products \(b_T:=\prod_\tau b_{T_\tau}(y_\tau)\) over the families of semistandard \(T_\tau\).  Changing the integral basis multiplies the coordinates by a matrix in \(\mathrm{GL}_{\dim W_\kappa}(\Z)\), which preserves the \(p\)-adic supremum norm and hence the gauge.

\begin{definition}\label{def:minors}
For a family \(y=(y_\tau)\) of complex \(n\times n\) matrices, the \emph{evaluation vector} \(\fl_\kappa(y)\in W_\kappa\) is \(F\mapsto F(y)\); its coordinates are the \(b_T(y)\).
\end{definition}

The action is \(\rho_\kappa(A)\fl_\kappa(y)=\prod_\tau\det(A_\tau)^{\kappa_{\tau,n}}\fl_\kappa((A_\tau y_\tau)_\tau)\).  The restriction of \(W_\kappa\) to \(\prod_\tau U(n)\) is irreducible, so its invariant positive definite hermitian form \(H\) is unique up to a positive factor.  We normalize \(H(\fl_\kappa(1),\fl_\kappa(1))=1\); the Weyl squared norm is \(\lvert v\rvert^2=H(v,v)\).  For \(\mathsf X\in\mathrm{GL}_n(E)\) and a ring map \(j\colon\C\to L\) to a field, \(\mathsf X^{(\tau)}\) is the entrywise image of \(\mathsf X\) under \(u\mapsto j(\Phi_\tau(\bar u))\).  For \(\mathsf X\in\mathrm{GL}_n(\mathbb A_{E,f})\) and \(j\colon\C\to\C_p\), take the component of \(\mathsf X\) at the finite place induced by this embedding of \(E\) and apply the continuous extension to the completion, as in Definition~\ref{ob-def:gauge}.

\begin{problem}\label{pr:sym2}
For \(n=2\), \(f=1\), \(\kappa=(2,0)\), the coordinates of \(\fl_\kappa(y)\) are \(y_{11}^2,y_{11}y_{21},y_{21}^2\), and the vectors at matrices with first columns \((1,1)\), \((1,-1)\), \((1,i)\) have determinant \(4\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:sym2

\begin{problem}\label{pr:eval-span}
The vectors \(\fl_\kappa(y)\), with all entries of all \(y_\tau\) roots of unity, span \(W_\kappa\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:eval-span

\begin{problem}\label{pr:eval-short}
Fix short bases, and read \(\mathsf X^{(\tau)}\) with \(j\) the identity.  For \(\tau\) let \(\B_\tau:=\B(g^{\kappa_{\tau,n}})\) be the short basis at the point \(g^{\kappa_{\tau,n}}\in G_f\) (not a power of \(\B(g)\)).  For \(y\) with root-of-unity entries put \(v_y:=\prod_\tau\det(\B_\tau^{(\tau)})\cdot\fl_\kappa\bigl((\B(g)^{(\tau)}y_\tau)_\tau\bigr)\).
\begin{enumerate}
\item[(a)] \(v_y\) has gauge at most \(1\) at \(g\) for every \(\eta\colon\C\to\C_p\) and every \(p\).
\item[(b)] \(\lvert\sigma v_y\rvert^2\le\csb^{6(nf+\lvert\kappa\rvert_{<n})}\) for every \(\sigma\in\Aut(\C)\).
\end{enumerate}
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:eval-short

\begin{problem}\label{pr:data-bound}
For the data, \(N'=q_1d\) lies in the different of \(E\), and with \(n=k\), \(4^f\csb^{6(kf+fa)}\le e^U\).
\end{problem}
% End problem: 20261126T080100--free-levels-and-short-bases__workbook.tex#pr:data-bound

The following theorem gives the short-spanning bound for a free level and its specialization to \(S\).

\begin{theorem}\label{thm:short-span}
\begin{enumerate}
\item[(a)] Let \(K\subseteq\Ko\) be a free level, \(\kappa\) a weight and \(Z\ge\csb^{6(nf+\lvert\kappa\rvert_{<n})}\).  The \(Z\)-short forms of weight \(\kappa\) and level \(K\) span the complex space of all forms of that weight and level.
\item[(b)] For data, the \(4^{-f}e^U\)-short forms in \(S\) span \(S\).
\end{enumerate}
\end{theorem}

Problems~\ref{pr:dim-free}, \ref{pr:delta}, \ref{pr:short-basis}, \ref{pr:eval-span} and~\ref{pr:eval-short} prove Theorem~\ref{thm:short-span}(a).  Problems~\ref{pr:free} and~\ref{pr:data-bound} give its specialization in~(b), which Chapter~\ref{ta-chap} uses to derive input~1 of Theorem~\ref{ta-thm:A}.  Problem~\ref{pr:pairing} gives the pairing condition of Definition~\ref{ob-def:target-lattice}.  Taken as given: transference \cite{Ba}; the basis of products of minors \cite{DEP}, \cite{Fu}; the Weyl module with its normalized invariant form; the integrality of the Weyl pairing, read from \cite{GL}; one class in the genus of \(\Z[i]^n\) for \(1\le n\le4\) (Problem~\ref{dt-pr:class-one}).

\end{description}
\subsection*{\S\ref{sec:eval}}
\begin{description}
\item[\ref{pr:eval-span}] Which polynomial corresponds to a linear functional that annihilates all these evaluations?
\item[\ref{pr:eval-short}] (a) At \(\eta\), compare the determinant factor with that of the integral matrix \(g^{-\kappa_{\tau,n}}\B_\tau\) under the same embedding.  (b) After applying \(\sigma\), use Hadamard's inequality on the determinants and a QR decomposition of each \(z_\tau=\sigma(\B(g)^{(\tau)}y_\tau)\).  If \(z_\tau=Q_\tau R_\tau\) with \(Q_\tau\) unitary and \(R_\tau\) upper triangular, which flag minors of \(R_\tau\) can be nonzero?
\item[\ref{pr:data-bound}] \(E\subseteq\Q(\zeta_{q_1d})\), whose different divides \(q_1d\).  Then \(f\le q_1\le b\), \(q_1d\le b\), \(k\le r\le a<b^2\) (Definition~\ref{da-def:data} and Problems~\ref{da-pr:chain} and~\ref{da-pr:m-big}) and \(\log b\ge 10\).
\end{description}
