\section{Odd classes}\label{sec:signs}

For odd primes \(\ell\) and odd \(e\ge1\), Theorem~\ref{tt-thm:MW} gives the length \(h_{\ell,e}(\Q)\) in terms of \(B_{e+1}/(e+1)\).  The proof \cite{GL} needs a bound of this kind over a CM field \(E\), for all \(\ell\) at once, and for cocycles with values in \(\Oc/\ell^n\).  Which module is bounded, and by what?

Recall from Section~\ref{tt-sec:odd} the CM field \(E\subset\overline\Q\), \(G_E=\Gal(\overline\Q/E)\), complex conjugation \(c\), \(\chi=\chi_\ell|_{G_E}\), the groups \(D_\Pp\supseteq I_\Pp\) in \(G_E\) of a place \(\Pp\) of \(\overline\Q\), and \(\Oc_n=\Oc/\ell^n\Oc\).  For \(e,n\ge0\), \(\Shao_\ell(E,\Oc;e,n)\) is the \(\Oc\)-module of maps \(z\colon G_E\to\Oc_n\) such that (Definition~\ref{tt-def:shaodd})
\begin{enumerate}
\item[(a)] \(z\) is locally constant and \(z(gh)=z(g)+\chi(g)^{-e}z(h)\) for all \(g,h\in G_E\);
\item[(b)] \(z=0\) on \(I_\Pp\) for every \(\Pp\) above a prime other than \(\ell\);
\item[(c)] for every \(\Pp\) above \(\ell\): \(z=0\) on \(I_\Pp\) if \(e=0\), and \(z=0\) on \(D_\Pp\cap\ker\chi\) if \(e>0\);
\item[(d)] if \(e>0\), then \(z(cgc)=-z(g)\) for all \(g\in G_E\).
\end{enumerate}
We write \(\So_{e,n}\) when \(\ell\), \(E\) and \(\Oc\) are fixed.  For \(e=0\), Theorem~\ref{tt-thm:e0} gives \(\len_\Oc\So_{0,n}\le v_\Oc(h_E)\) for every CM field \(E\).  This chapter treats \(e>0\).

\begin{problem}\label{pr:split}
Let \(R\) be a commutative ring with \(2\in R^\times\), \(M\) an \(R\)-module of finite length, and \(\sigma\) an \(R\)-linear involution of \(M\).  Then \(M=M^{\sigma=1}\oplus M^{\sigma=-1}\) and \(\len M=\len M^{\sigma=1}+\len M^{\sigma=-1}\).
\end{problem}
% End problem: 236W

By Problem~\ref{tt-pr:c-action}, for odd \(\ell\) and \(e>0\) the module of maps satisfying (a)--(c) therefore splits, when it has finite length, into \(\So_{e,n}\) and the part where \(z(cgc)=z(g)\).  Over \(\Q\) the odd cocycles are seen through characters.  Let \(e>0\) and \(E\subseteq\Q(\mu_N)\); in this chapter \(N\) is such a modulus, as in Theorem~\ref{tt-thm:oddsha}, not the level of Chapter~\ref{da-chap}.  Suppose that \(\Oc\) contains the values of all characters modulo \(N\).  Write \(H=\Gal(\overline\Q/\Q(\mu_N))\).  For an \(\Oc\)-valued Dirichlet character \(\psi\) modulo \(N\), put \(\Upsilon=\chi_\ell^e\cdot(\psi\circ\chi_N)\), where \(\chi_N\) is the cyclotomic character modulo \(N\).  We define the \emph{component} of \(\psi\) to consist of the locally constant \(\Upsilon^{-1}\)-cocycles \(\Gal(\overline\Q/\Q)\to\Oc_n\) that vanish at \(c\) and at the elements of \(H\) that lie in an inertia group at a place away from \(\ell\), or in a decomposition group above \(\ell\) with \(\chi_\ell(g)=1\); here inertia and decomposition groups are taken in \(\Gal(\overline\Q/\Q)\).  The proof associates to each \(z\in\So_{e,n}\) a cocycle in each component; if the associated cocycles vanish for every \(\psi\) with \(\Upsilon(c)=-1\), then \(4\varphi(N)z=0\).

\begin{problem}\label{pr:parts}
\(\chi_\ell(c)=-1\).  Hence \(\Upsilon(c)=-1\) if and only if \(\psi(-1)=(-1)^{e+1}\), that is, if and only if \(\psi\) is \emph{critical} for \(e\).
\end{problem}
% End problem: 236X

\begin{problem}\label{pr:parity}
Let \(\chi\) be a primitive Dirichlet character and \(k\ge2\).  Then \(B_{k,\chi}=0\) if \(\chi(-1)\ne(-1)^k\), and \(B_{k,\chi}\ne0\) if \(\chi(-1)=(-1)^k\).  So \(L(-e,\chi)\ne0\), for \(e\ge1\), exactly when \(\chi\) is critical for \(e\).
\end{problem}
% End problem: 236Y

The next problem describes the kind of field used in the proof.

\begin{problem}\label{pr:degree}
Let \(d\equiv3\pmod4\) be prime and \(q\ge1\).  Then \(\Q(\sqrt{-d})\subseteq\Q(\mu_d)\), and \(\Q(\sqrt{-d},\mu_q)\) is a CM field contained in \(\Q(\mu_{qd})\).
\end{problem}
% End problem: 236Z

The proof specializes the theorem of this chapter to \(E=\Q(\sqrt{-d},\mu_{q_1})\), \(N=q_1d\), and the integer \(t\) of its admissible data; the chosen embedding of \(E\) has image in \(\Q(\mu_N)\) \cite{GL}.

\section{One integer for all characters}\label{sec:bernoulli}

On the components of the critical characters an \(L\)-value is available.  How can one integer serve for all the characters modulo \(N\) at once?  Fix \(N,e\ge1\); for \(N=1\) the character modulo \(1\) takes the value \(1\) at \(0\).  For \(0\le a<N\) put
\[
 \beta_a=(e+2)!\,N^{e+1}B_{e+1}(a/N).
\]

\begin{problem}\label{pr:integral}
For integers \(k\ge1\), \(N\ge1\) and \(a\), the number \((k+1)!\,N^kB_k(a/N)\) is an integer.  In particular \(\beta_a\in\Z\).
\end{problem}
% End problem: 2373

\begin{problem}\label{pr:hurwitz}
Let \(\chi\) be a Dirichlet character modulo \(N\), not necessarily primitive.  Then \(\sum_{0\le a<N}\chi(a)\beta_a=-(e+2)!\,(e+1)\,N\,L(-e,\chi)\).  For \(N=3\), \(e=1\) and \(\chi\) trivial both sides are \(-6\).
\end{problem}
% End problem: 2374

\begin{problem}\label{pr:size}
For \(k\ge2\) and \(0\le s\le1\), \(\lvert B_k(s)\rvert\le4\,k!/(2\pi)^k\).  Hence \(\lvert\beta_a\rvert<(e+2)!\,(e+1)!\,N^{e+1}\).
\end{problem}
% End problem: 2375

Let \(U_N=(\Z/N)^\times\).  For \(t\in U_N\) put \(\kappa(t)=2\beta_t+(-1)^e[t=1]+[t=-1]\), where \(\beta_t\) is taken at the representative of \(t\) in \([0,N)\) and \([\,\cdot\,]\) is \(1\) or \(0\) as the condition holds or not.  Let \(T\) be the integer matrix with entries \(T_{s,t}=\kappa(t/s)\), \(s,t\in U_N\).  In this chapter \(T\) and \(\kappa\) denote this matrix and this function.

\begin{problem}\label{pr:matrix}
For each Dirichlet character \(\psi\) modulo \(N\), the vector \((\psi(t))_{t\in U_N}\) is an eigenvector of \(T\) with eigenvalue \(\lambda_\psi=\sum_{t\in U_N}\psi(t)\kappa(t)\), and \(\det T=\prod_\psi\lambda_\psi\).
\end{problem}
% End problem: 2376

\begin{problem}\label{pr:eigen}
\(\lambda_\psi=2\sum_{0\le a<N}\psi(a)\beta_a+(-1)^e+\psi(-1)\).  It equals \(2\sum_a\psi(a)\beta_a\ne0\) if \(\psi\) is critical for \(e\), and \(2(-1)^e\) otherwise.  In particular \(\det T\ne0\).
\end{problem}
% End problem: 2377

The correction \((-1)^e[t=1]+[t=-1]\) does nothing on the critical characters and replaces the zeros of Problem~\ref{pr:parity} by \(\pm2\).  One integer is divisible by every eigenvalue.

\begin{problem}\label{pr:adjugate}
Let \(R\) be a commutative ring, \(A\) a commutative \(R\)-algebra, free of finite rank as an \(R\)-module, \(z\in A\), and \(\Delta=\det_R(z\mid A)\).  Then \(\Delta\in zA\).  Hence every \(A\)-module killed by \(z\) is killed by \(\Delta\), and, with \(A=\Z[U_N]\) and \(z=\sum_t\kappa(t)\,t\), every \(\lambda_\psi\) divides \(\det T\) in \(\Z[\psi]\).
\end{problem}
% End problem: 2378

\begin{problem}\label{pr:simplest}
For \(N=3\) and \(e=1\): \(\beta_1=\beta_2=-3\), \(\kappa(1)=-7\), \(\kappa(2)=-5\), \(\lambda_\psi=-12\) for the trivial character, which is critical, \(\lambda_\psi=-2\) for the other, and \(\det T=24\).
\end{problem}
% End problem: 2379

The eigenvalues are tied to Stickelberger sums by a character-weighted form of Voronoi's congruence, as in the proof \cite{GL}; for the classical congruence see \cite[Ch.~15, \S2]{IR}.

\begin{problem}\label{pr:zcong}
Let \(u\ge0\) be an integer with \(u\equiv1\pmod N\), let \(F\ge1\) be prime to \(u\), and let \(\chi\) be a Dirichlet character modulo \(N\).  Then, in \(\Z[\chi]\),
\[
 (e+2)!\,(e+1)\,N\,u^e\sum_{0\le a<NF}\Bigl\lfloor\frac{ua}{NF}\Bigr\rfloor a^e\chi(a)\equiv(u^{e+1}-1)\sum_{0\le a<N}\chi(a)\beta_a\pmod F.
\]
For \(N=1\), \(e=1\), \(u=2\), \(F=5\) the two sides are \(168\) and \(3\); for \(N=3\), \(e=1\), \(u=4\), \(F=5\) and \(\chi\) trivial they are \(22320\) and \(-90\).
\end{problem}
% End problem: 237A

\begin{problem}\label{pr:witness}
Let \(\psi\) be critical for \(e\), let \(X\) be an integer divisible by \(\det T\), and let \(u\ge0\), \(u\equiv1\pmod N\).  There is \(w\in\Z[\psi]\) such that for every \(F\ge1\) prime to \(u\),
\[
 (u^{e+1}-1)X\equiv2w\sum_{0\le a<NF}\Bigl\lfloor\frac{ua}{NF}\Bigr\rfloor a^e\psi(a)\pmod F.
\]
\end{problem}
% End problem: 237B

For fixed \(N,e\ge1\), the proof uses the integer
\[
 X=\bigl((1+N)^e-1\bigr)\bigl((1+N)^{e+1}-1\bigr)\bigl((1+2N)^e-1\bigr)\bigl((1+2N)^{e+1}-1\bigr)\,\lvert\det T\rvert,
\]
so that for every prime \(\ell\) one of \(u=1+N\), \(u=1+2N\) is prime to \(\ell\) and \((u^e-1)(u^{e+1}-1)\mid X\).  It bounds \(\lvert\det T\rvert\) through the entries of \(T\), using \(\lvert B_{e+1}(s)\rvert<17\,(e+1)!\) on \([0,1]\), and obtains the following.

\begin{theorem}[Stickelberger bound]\label{thm:stick}
Let \(N,e\ge1\).  There is an integer \(X\) with \(1\le X\le2^{(N+e+3)^6}\) such that
\begin{enumerate}
\item[(a)] for every prime \(\ell\) there is an integer \(u\ge1\) with \(u\equiv1\pmod N\), \(\ell\nmid u\) and \((u^e-1)(u^{e+1}-1)\mid X\);
\item[(b)] for every integer \(u\ge0\) with \(u\equiv1\pmod N\), every integral domain \(\Lambda\) of characteristic \(0\), and every Dirichlet character \(\psi\) modulo \(N\) with values in \(\Lambda\) and \(\psi(-1)=-(-1)^e\), there is \(w\in\Lambda\) such that for every integer \(F\ge1\) prime to \(u\), in \(\Lambda\),
\[
 F\ \Big|\ (u^{e+1}-1)X-2w\sum_{0\le a<NF}\Bigl\lfloor\frac{ua}{NF}\Bigr\rfloor a^e\psi(a).
\]
\end{enumerate}
\end{theorem}

For \(N=3\) and \(e=1\) this \(X\) is \(311040\).  

\section{Gauss sums and the Euler system}\label{sec:euler}

How does an integer such as \(X\) bound a module of cocycles from above, without the main conjecture?  The proof uses Kolyvagin's method for Gauss sums, in Rubin's version for class groups \cite{Ru91}, \cite[Appendix]{La}, \cite{Gr}, with a Tate twist as in \cite{BG}.  At level \(\ell^n\), let \(u\ge1\) be an auxiliary exponent, \(P\) a finite set of primes \(q\equiv1\pmod{uN\ell^n}\), and \(p\) a prime congruent to \(1\) modulo \(N\ell^n\prod_{q\in P}q\).  The Gauss sums at \(p\) give elements of known valuation.  For \(S\subseteq P\) the derivative cocycle \(\kappa_S\) has character \(\chi_\ell^{e+1}(\psi\circ\chi_N)\); over \(\Q(\mu_N)\) it is unramified outside \(\ell\), \(p\) and \(S\); and at \(q\in P\setminus S\) its value at a Frobenius is a value of \(\kappa_{S\cup\{q\}}\) on inertia at the same place.

For \(x\in\Q/\Z\) let \(\langle x\rangle\in[0,1)\) be its representative, and for an integer \(u\ge0\) put
\[
 \vartheta_u(x)=\lfloor u\langle x\rangle\rfloor-\lfloor u\langle -x\rangle\rfloor .
\]
For \(F\ge2\) let \(\theta_F=\sum_a\langle a/F\rangle\,\sigma_a^{-1}\in\Q[(\Z/F)^\times]\), the sum over \(a\in(\Z/F)^\times\), be the Stickelberger element at modulus \(F\), extending the prime-modulus definition of Chapter~\ref{ds-chap}.

\begin{problem}\label{pr:theta}
Let \(u\ge1\) be prime to \(F\).  For \(a\) prime to \(F\), \(\vartheta_u(a/F)=2\lfloor u\langle a/F\rangle\rfloor-u+1\), and \(\sum_a\vartheta_u(a/F)\,\sigma_a^{-1}=2(u-\sigma_u)\theta_F-(u-1)\sum_a\sigma_a\).  For \(F=5\), \(u=2\) the values at \(a=1,2,3,4\) are \(-1,-1,1,1\).
\end{problem}
% End problem: 237E

Fix a place \(\Pp\) of \(\overline{\Q}\) above \(p\), normalize its valuation by \(v_\Pp(p)=1\), and form all Gauss sums \(g(\chi)\) with the same nontrivial additive character of \(\F_p\).  By Stickelberger's factorization \cite[Ch.~6]{Wa}, the characters \(\chi\) of \(\F_p^\times\) can be indexed by the points \(x\in\Q/\Z\) of order dividing \(p-1\) so that \(j(\chi)=g(\chi)^u/g(\chi^u)\) satisfies \(v_\Pp(j(\chi))=\lfloor u\langle x\rangle\rfloor\) and \(v_\Pp(j(\chi)/j(\chi^{-1}))=\vartheta_u(x)\).  The proof builds its elements of known valuation from these quotients.  The next problem is a cyclic group-ring model for the derivative operator.

\begin{problem}\label{pr:derivative}
Let \(\sigma\) generate a cyclic group of order \(M\), \(D=\sum_{i=1}^{M-1}i\sigma^i\) and \(\mathcal N=\sum_{i=0}^{M-1}\sigma^i\).  Then \((\sigma-1)D=M-\mathcal N\).
\end{problem}
% End problem: 237F

The Hilbert symbol over \(\Q\) gives a classical example of global reciprocity \cite[Ch.~III]{Se}.

\begin{problem}\label{pr:hilbert}
The Hilbert symbol \((-1,-1)_v\) is \(-1\) for \(v=\infty\) and \(v=2\), and \(1\) for odd \(v\).  If \(a,b\in\Q^\times\) and \((a,b)_v=1\) for all places \(v\ne r\), then \((a,b)_r=1\).
\end{problem}
% End problem: 237G

The reciprocity law used in the proof is Artin reciprocity over \(\Q(\mu_{Nm})\) at primes \(q\equiv1\pmod{Nm}\), where \(m\ge1\) kills the component cocycle.  Let \(e\ge1\), \(\psi\) critical for \(e\), and \(X\) as in Theorem~\ref{thm:stick}.  Then \(4X^2\) kills the component of \(\psi\), an analogue of Stickelberger's theorem; and if \(\ell>N+e+3\), Kolyvagin's descent gives a family \((\gamma_i)_{0\le i\le v_\ell(X)}\) in \(\Gal(\overline\Q/\Q)\) such that every cocycle in the component of \(\psi\) vanishing at all the \(\gamma_i\) is zero.  For a CM field \(E\subseteq\Q(\mu_N)\) and \(\Oc\) containing the values of the characters modulo \(N\), it follows that \(Y_e=4\varphi(N)\cdot4X^2\) kills \(\So_{e,n}\), and that for \(\ell>y^2\), where \(y\ge N+e+3\), the elements of \(G_E\) obtained from triples (a unit modulo \(N\), a character \(\psi\), an index \(0\le i\le v_\ell(X)\)) detect \(\So_{e,n}\).  For each \(E\subseteq\Q(\mu_N)\) and each prime \(\ell\), there is a finite set in \(G_E\) detecting all maps satisfying (a) and (b), simultaneously for all \(\Oc\), \(n\), \(e\); for every integer \(y\ge N\) with \(\ell\le y^2\), it has at most \(y^{20}\) elements.

\begin{problem}\label{pr:exponent-rank}
Let \(Z\) be an \(\Oc\)-submodule of the functions \(G_E\to\Oc_n\), killed by an integer \(m\ge1\), and let \(g_1,\dots,g_k\in G_E\) be such that the only \(z\in Z\) with \(z(g_1)=\dots=z(g_k)=0\) is \(z=0\).  Then \(\len_\Oc Z\le v_\Oc(m^k)\).
\end{problem}
% End problem: 237H

If \(X\le2^{y^6}\), \(y\ge4\) and \(N\le y\), there are at most \(\varphi(N)^2(y^6+1)\le y^{20}\) triples.  So for \(0<e<t\), every \(\ell\) and \(y=N+t+3\), \(\len_\Oc\So_{e,n}\le v_\Oc(Y_e^{y^{20}})\), where \(X_e\) is an integer as in Theorem~\ref{thm:stick} for \(N\) and \(e\), and \(Y_e=4\varphi(N)\cdot4X_e^2\).  One integer for all \(0<e<t\) is \(\bigl(\prod_{0<e<t}Y_e\bigr)^{y^{20}}\).

\begin{problem}\label{pr:sum-e}
Let \(N\ge1\), \(t\ge0\), \(y=N+t+3\), and let \(1\le X_e\le2^{(N+e+3)^6}\) for \(0<e<t\).  Then
\[
 \Bigl(\prod_{0<e<t}4\varphi(N)\cdot4X_e^2\Bigr)^{y^{20}}\le2^{4y^{27}}.
\]
\end{problem}
% End problem: 237J

Theorem~\ref{tt-thm:e0}, Problems~\ref{pr:exponent-rank} and~\ref{pr:sum-e}, Theorem~\ref{thm:stick} and the statements above prove Theorem~\ref{tt-thm:oddsha}, with \(X_{\mathrm{odd}}=\bigl(\prod_{0<e<t}Y_e\bigr)^{y^{20}}h_E\): one integer \(X_{\mathrm{odd}}\le2^{4(N+t+3)^{27}}h_E\) bounds \(\len_\Oc\Shao_\ell(E,\Oc;e,n)\) by \(v_\Oc(X_{\mathrm{odd}})\) for every prime \(\ell\), \(2\) and the primes dividing \(N\) included, every \(\Oc\) as in that theorem, every \(0\le e<t\) and every \(n\ge0\).

\section{Examples, and the main conjecture}\label{sec:examples}

How much is lost?  For odd \(\ell\) and odd \(e\ge1\), we can compare the eigenvalue for the principal character modulo \(\ell\) with the exact value of \(h_{\ell,e}(\Q)\) in Theorem~\ref{tt-thm:MW}.  First we compute the \(37\)-adic valuations of the eigenvalues for all critical characters modulo \(37\) at \(e=1\).  For \(N>1\) and \(\psi\) critical, \(\lambda_\psi=2\,(e+2)!\,N\,B^{(N)}_{e+1,\psi}\) with \(B^{(N)}_{k,\psi}=N^{k-1}\sum_{a=1}^N\psi(a)B_k(a/N)\), which is \(B_{k,\psi}\) when \(\psi\) is primitive of conductor \(N\).  Let \(\omega\) be the Teichm\"uller character modulo \(\ell\).

\begin{problem}\label{pr:37}
Let \(\ell=N=37\), \(e=1\), and \(\psi=\omega^j\) with \(j\) even, \(0\le j<36\).  Then \(v_{37}(\lambda_\psi)=1+v_{37}(B_{2,\omega^j})\) for \(j\ne0\), and \(v_{37}(\lambda_\psi)=1\) for \(j=0\).  It equals \(2\) for \(j=30\), \(0\) for \(j=34\), and \(1\) for the other \(j\).
\end{problem}
% End problem: 237L

At \(j=30\) the value reflects the irregular pair \((37,32)\) of Problem~\ref{ds-pr:irregular}, through Kummer's congruences \cite[Ch.~5]{Wa}.

\begin{problem}\label{pr:psi-one}
Let \(\ell\) be odd, \(N=\ell\), \(\psi=1\) and \(e\) odd.  Then \(v_\ell(\lambda_\psi)=v_\ell(B_{e+1})+1+v_\ell\bigl((e+2)!\bigr)\).  For \(\ell=691\), \(e=11\) this is \(2\), while \(h_{691,11}(\Q)=1\).
\end{problem}
% End problem: 237M

\begin{problem}\label{pr:compare}
If \(\ell\) is odd, \(e\) is odd and \((\ell-1)\nmid(e+1)\) (so \(\ell\ge5\)), the value of Problem~\ref{pr:psi-one} exceeds the value \(h_{\ell,e}(\Q)\) given by Theorem~\ref{tt-thm:MW} by exactly \(1+v_\ell\bigl((e+2)!\bigr)+v_\ell(e+1)\).
\end{problem}
% End problem: 237N

For each \(0<e<t\), one may take \(X_e\) to be the determinant multiple of Section~\ref{sec:bernoulli}; each eigenvalue for that \(e\) divides \(X_e\) in its character ring, and the witness above contains the squares \(X_e^2\) in a product raised to the power \(y^{20}\).  Why is so coarse a bound good enough, and why does a bound at finite level replace the main conjecture?  The proof applies Theorem~\ref{tt-thm:oddsha} to choose a witness \(X_{\mathrm{odd}}\) for the data of Chapter~\ref{da-chap}.  At every prime \(\ell\), \(v_\Oc(X_{\mathrm{odd}})\) bounds the lengths in that theorem for all sufficiently large coefficient rings \(\Oc\) in the tower of finite Galois subextensions of \(\overline{\Q}_\ell/\Ql\), ordered by inclusion.  The denominator error at \(\ell\) includes \(t\,v_\ell(X_{\mathrm{odd}})\) (Chapter~\ref{dp-chap}) \cite{GL}.  The main conjecture of Mazur and Wiles \cite{MW} gives the formula of Theorem~\ref{tt-thm:MW} for the rational Tate twists; the proof notes that for large \(\ell\) a bound of this kind follows from it, and proves only an upper bound at finite level.  To bound the valuations \(v_\ell(X_{\mathrm{odd}})\) with weights \(\log\ell\) it uses the size of \(X_{\mathrm{odd}}\), by Problem~\ref{tt-pr:one-number}:
\[
 \sum_{\ell\in\mathcal L}v_\ell(X_{\mathrm{odd}})\log\ell\le\log X_{\mathrm{odd}}\le4(N+t+3)^{27}\log2+\log h_E
\]
for every finite set \(\mathcal L\) of primes.  For the data, \(\log h_E\le x^{30}\), the sum is at most \(x^{2000}\), and \(x^{10^6}\le x_1\), \(x_1^{10^6}\le b\) \cite{GL}.  So the sum is at most \(b^{1/(5\cdot10^8)}\), and its contribution to the denominator errors, weighted by \(\log\ell\), is at most \(t\,x^{2000}\).

\paragraph{In the Lean formalization.}
In the Lean proof \cite{GL}, Theorem~\ref{tt-thm:oddsha} is \tl{New.\allowbreak{}exists\_\allowbreak{}length\_\allowbreak{}shaOdd\_\allowbreak{}le}, in \tl{Lib/\allowbreak{}Cyclotomic/\allowbreak{}OddSha.lean}, and Theorem~\ref{thm:stick} is \tl{New.\allowbreak{}OddPart.\allowbreak{}exists\_\allowbreak{}isStickelbergerBound}, in \tl{Lib/\allowbreak{}Cyclotomic/\allowbreak{}Voronoi.lean}.

\item[\ref{pr:integral}] Expand \(B_k(a/N)\) in powers of \(a/N\), and apply von Staudt--Clausen to the coefficients.
\item[\ref{pr:hurwitz}] \(L(s,\chi)=N^{-s}\sum_{a=1}^N\chi(a)\zeta(s,a/N)\) and \(\zeta(-e,x)=-B_{e+1}(x)/(e+1)\) \cite[Ch.~4]{Wa}.
\item[\ref{pr:size}] The Fourier series of \(B_k(s)\) on \([0,1]\), obtained by integrating by parts with \(B_k'=kB_{k-1}\); and \(\zeta(k)\le\zeta(2)\).
\item[\ref{pr:matrix}] Substitute \(t=st'\); then orthogonality of characters.
\item[\ref{pr:eigen}] Problems~\ref{pr:hurwitz} and~\ref{pr:parity}, and \(L(-e,\chi)=L(-e,\chi_0)\prod_{p\mid N}(1-\chi_0(p)p^e)\) for the primitive \(\chi_0\) inducing \(\chi\).
\item[\ref{pr:adjugate}] The Cayley--Hamilton theorem for multiplication by \(z\); in the basis \(U_N\) its matrix is the transpose of \(T\).
\item[\ref{pr:zcong}] Write \(ua=r_a+NF q_a\) with \(0\le r_a<NF\), expand \((ua)^{e+1}\) modulo \((NF)^2\) and sum against \(\chi(a)\); \(a\mapsto r_a\) permutes the residues.  Evaluate \(\sum_{0\le a<NF}\chi(a)a^{e+1}\) by \(\sum_{0\le i<n}(x+i)^k=\bigl(B_{k+1}(x+n)-B_{k+1}(x)\bigr)/(k+1)\), and control denominators with Problem~\ref{pr:integral}.
\item[\ref{pr:witness}] Problems~\ref{pr:adjugate} and~\ref{pr:zcong}.
\end{description}

\noindent\S\ref{sec:euler}.
\begin{description}\setlength{\itemsep}{0pt}
\item[\ref{pr:theta}] Here \(F\nmid ua\).  For the group-ring identity, reindex the sum over units as in the prime-modulus Problem~\ref{ds-pr:c-twist}.
\item[\ref{pr:derivative}] Telescope, using \(\sigma^M=1\).
\item[\ref{pr:hilbert}] The formulas for \((a,b)_v\) and the product formula \cite[Ch.~III, \S2]{Se}.
\item[\ref{pr:exponent-rank}] Evaluate at \(g_1,\dots,g_k\).
\item[\ref{pr:sum-e}] Bound one factor by a power of \(2\).
\end{description}

\noindent\S\ref{sec:examples}.
\begin{description}\setlength{\itemsep}{0pt}
\item[\ref{pr:37}] Compute \(\lambda_\psi=2\sum_{t=1}^{36}\omega(t)^j(6t^2-222t+1369)\) in \(\Z_{37}\) to a few digits.
\item[\ref{pr:psi-one}] \(B^{(\ell)}_{k,1}=(1-\ell^{k-1})B_k\), \(B_{12}=-691/2730\), and Theorem~\ref{tt-thm:MW}.
\item[\ref{pr:compare}] \(B_{e+1}/(e+1)\) is \(\ell\)-integral when \((\ell-1)\nmid(e+1)\) \cite[Cor.~5.14]{Wa}.
\end{description}
