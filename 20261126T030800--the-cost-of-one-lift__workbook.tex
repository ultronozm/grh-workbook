\paragraph{Prerequisites.}
Linear algebra over \(\C\): adjoints, orthogonal projections, determinants as products of eigenvalues.  The fields \(\bar\Q_\ell\) and \(\C_\ell\), with \(\lvert\ell\rvert_\ell = 1/\ell\), and the product formula for a number field.  From Chapter~\ref{c28-chap}: congruence modules, Problems~\ref{c28-pr:symmetry} and~\ref{c28-pr:denominator}.  From Chapter~\ref{c22-chap}: Problem~\ref{c22-pr:U1U1} and Theorem~\ref{c22-thm:norm}.  From Chapter~\ref{c6-chap}: sizes read at the trivial twist, valuations at a twist that depends on the prime.  For Section~\ref{sec:general} only: the data of Chapter~\ref{c32-chap} and the single lifts of the chain in Chapter~\ref{c37-chap}.

\medskip
Chapter~\ref{c37-chap} reduces the Gram determinant of the chain to the Gram determinants \(Z_j(\xi)\) of single lifts, where for an \(m\)-fold twist \(\xi\) the \(j\)-th lift is taken at the twist \(\xi^j\).  The proof needs a bound for the height of \(Z_j(1)/Z_j(\xi)\) when \(\xi\) varies with the prime \(\ell\) and with an isomorphism \(\C \simeq \bar\Q_\ell\).  How does the rewritten proof bound the ratio without evaluating either determinant individually?  One multiplies the Gram determinant at a twist by the determinant of the compression of a second operator to the image of the lift, chosen so that the product is a fixed constant times a power of a ratio of Hecke values.  The compression determinant is then bounded under the automorphisms of \(\C\) at the trivial twist only, and at \(\ell\) at the twist.  We first do this for lines in \(\C^3\), then state the general theorem in the form of the rewritten Lean proof \cite{GL}.

\section{Compressions}\label{sec:compress}

Let \(A\) and \(V\) be finite-dimensional complex inner product spaces, with inner products conjugate-linear in the first variable, and let \(u \colon A \to V\) be injective, with image \(R\).  Write \(\pr_R\) for the orthogonal projection of \(V\) onto \(R\).  For \(T \in \End V\) the \emph{compression} of \(T\) to \(R\) is \(\pr_R T|_R \in \End R\).

\begin{problem}\label{pr:compression}
For every \(T \in \End V\),
\[
  \det(u^*u)\,\det(\pr_R T|_R) = \det(u^* T u) .
\]
\end{problem}

Now let \(\vartheta\) run over a set containing an element \(1\), and suppose given for each \(\vartheta\) an injective \(u_\vartheta \colon A \to V_\vartheta\) with image \(R_\vartheta\), an operator \(U_\vartheta \in \End V_\vartheta\), and a complex number \(\rho_\vartheta \ne 0\).  Put \(Z(\vartheta) := \det(u_\vartheta^* u_\vartheta)\) and \(\Delta_\vartheta := \det(\pr_{R_\vartheta} U_\vartheta|_{R_\vartheta})\).

\begin{problem}\label{pr:comparison}
Suppose that some \(U_0 \in \End A\) has \(\langle y, U_0 y\rangle > 0\) for every \(y \ne 0\) and \(\langle u_\vartheta y', U_\vartheta u_\vartheta y\rangle = \rho_\vartheta \langle y', U_0 y\rangle\) for all \(\vartheta\) and \(y', y \in A\).  Then \(\det U_0 \ne 0\), every \(\Delta_\vartheta \ne 0\), and
\[
  Z(\vartheta)\,\Delta_\vartheta = \det(U_0)\, \rho_\vartheta^{\dim A}
  \qquad\text{for every } \vartheta .
\]
If moreover \(\rho_1 = 1\), then \(Z(1)/Z(\vartheta) = (\Delta_\vartheta/\Delta_1)\,\rho_\vartheta^{-\dim A}\).
\end{problem}

So the unknown \(Z(\vartheta)\) is traded for \(\rho_\vartheta\) and for \(\Delta_\vartheta\), the determinant of an operator on the image.  The simplest such operators have rank one.

\begin{problem}\label{pr:rank-one}
Let \(A = \C\), \(u_\vartheta(1) = v_\vartheta\), and \(U_\vartheta = g_\vartheta g_\vartheta^*\) with \(g_\vartheta \colon \C \to V_\vartheta\), \(g_\vartheta(1) = w_\vartheta\).  Then \(Z(\vartheta) = \lVert v_\vartheta\rVert^2\) and \(\Delta_\vartheta = \lvert\langle w_\vartheta, v_\vartheta\rangle\rvert^2/\lVert v_\vartheta\rVert^2\).  If \(\langle w_\vartheta, v_\vartheta\rangle = c_\vartheta^{-1} P\) with \(c_\vartheta \ne 0\) and \(P \ne 0\) independent of \(\vartheta\), the hypothesis of Problem~\ref{pr:comparison} holds with \(\rho_\vartheta = \lvert c_1\rvert^2/\lvert c_\vartheta\rvert^2\) and \(U_0 = \lvert P\rvert^2/\lvert c_1\rvert^2\).
\end{problem}

For instance, if \(4 \mid a\) and \(c_a \ne 0\) (as for \(a = 0\)), the lift \(\Theta_{(a)}\) of Chapter~\ref{c22-chap} between two lines is multiplication by \(c_a/4\) (Problem~\ref{c22-pr:U1U1}); so \(A = V = \C\), and \(Z = c_a^2/16\) by Theorem~\ref{c22-thm:norm}.

\begin{problem}\label{pr:eigen}
Let \(L\) be a finite-dimensional complex vector space, \(T \in \End L\), \(f \colon \C \to F\) a ring homomorphism into a field with an absolute value, and \(B \in \R\).  If \(\lvert f(\alpha)\rvert \le B\) for every eigenvalue \(\alpha\) of \(T\), then \(\lvert f(\det T)\rvert \le B^{\dim L}\).
\end{problem}

\section{A line in \texorpdfstring{\(\C^3\)}{C3}}\label{sec:toy}

Where do the bounds for \(\Delta_\vartheta\) come from?  Let \(V = \C^n\) with \(\langle y, z\rangle = \sum \bar y_i z_i\).  An automorphism \(\sigma\) of \(\C\) acts on \(V\) coordinatewise.  For a prime \(\ell\) and a field isomorphism \(\iota \colon \C \to \bar\Q_\ell\) let \(\Lc_\iota := \{y \in V : \lvert\iota(y_i)\rvert_\ell \le 1 \text{ for all } i\}\), the vectors \emph{integral at} \(\iota\).

Let \(K \subset \C\) be a subfield that is Galois over \(\Q\) with abelian Galois group, such as \(\Q\) or \(\Q(\ii)\), and let \(X \subseteq V\) be spanned by vectors with coordinates in \(K\); in Section~\ref{sec:general} the corresponding space is the intersection of the image of an idempotent with the kernel of a Hecke operator.  Let \(R \subseteq X\) be a subspace, the \emph{lifts}, which need not be spanned by vectors over \(K\), and let \(F := X \cap R^\perp\), the \emph{false lifts}: they satisfy the condition that defines \(X\) but are orthogonal to the lifts.  Let \(U \in \End V\) with \(U(R) \perp F\), and put \(\Delta := \det(\pr_R U|_R)\).

\begin{problem}\label{pr:false}
If \(y \perp F\), then \(\pr_R y = \pr_X y\).
\end{problem}

\begin{problem}\label{pr:conj}
For \(\sigma \in \Aut(\C)\) and \(y \in V\), \(\sigma(\pr_X y) = \pr_{\sigma X}(\sigma y)\), where \(\sigma X\) is spanned by the images under \(\sigma\) of the spanning vectors with coordinates in \(K\) (it does not depend on their choice).  Hence \(\lVert\sigma(\pr_R y)\rVert \le \lVert\sigma y\rVert\) whenever \(y \perp F\).
\end{problem}

\begin{problem}\label{pr:arch}
If \(\lVert\sigma(U\varphi)\rVert \le e^{c}\lVert\sigma\varphi\rVert\) for all \(\varphi \in V\) and \(\sigma \in \Aut(\C)\), then \(\lvert\sigma(\Delta)\rvert \le e^{c\dim R}\) for all \(\sigma\).
\end{problem}

\begin{problem}\label{pr:padic}
Suppose that \(U(\Lc_\iota) \subseteq \Lc_\iota\), and that \(\Lambda \in \End R\) satisfies \(\Lambda(\pr_R y) \in \Lc_\iota\) for every \(y \in \Lc_\iota\) with \(y \perp F\).  Then \(\lvert\iota(\det\Lambda \cdot \Delta)\rvert_\ell \le 1\).
\end{problem}

\begin{problem}\label{pr:clear}
Let \(K = \Q\), and let \(c\) be a positive integer with \(c \pr_X(\Z^n) \subseteq \Z^n\).  At every \(\iota\), the operator \(\Lambda = c \cdot \mathrm{id}_R\) satisfies \(\Lambda(\pr_R y) \in \Lc_\iota\) for every \(y \in \Lc_\iota\) with \(y \perp F\).  If moreover \(U(\Lc_\iota) \subseteq \Lc_\iota\), then \(\lvert\iota(\Delta)\rvert_\ell \le \lvert c\rvert_\ell^{-\dim R}\).
\end{problem}

By Problem~\ref{c28-pr:denominator}, the least such \(c\) is the exponent of the congruence module of \(X\) in \(\Z^n\) for the decomposition \(\Q^n = (X \cap \Q^n) \oplus (X^\perp \cap \Q^n)\).  So the denominator of \(\Delta\) is bounded by a congruence module, and its conjugates by the size of \(U\); neither bound uses \(\det(u^*u)\).  Here is a three-dimensional example with a denominator, in which the lifts form a line inside a plane.  Let \(n = 3\) and \(X := \{y : y_1 + y_2 + y_3 = 0\}\).

\begin{problem}\label{pr:toy-cong}
The congruence module of \(X\) in \(\Z^3\) is \(\Z/3\Z\), and \(3\pr_X(\Z^3) \subseteq \Z^3\).
\end{problem}

\begin{problem}\label{pr:toy-delta}
Let \(u_1(z) := zv_1\) and \(u_\vartheta(z) := zv_\vartheta\) for \(z \in \C\), with \(v_1 := (1,-1,0)\) and \(v_\vartheta := (1,1,-2)\), so that \(R_1 = \C v_1\) and \(R_\vartheta = \C v_\vartheta\); and let \(U_1 := w_1 w_1^*\) and \(U_\vartheta := w_\vartheta w_\vartheta^*\) with \(w_1 := (2,0,1)\) and \(w_\vartheta := (1,1,0)\).  Then \(U_1(R_1)\) and \(U_\vartheta(R_\vartheta)\) are orthogonal to the respective false lifts, \(\Delta_1 = 2\) and \(\Delta_\vartheta = 2/3\), and \(Z(1)\Delta_1 = Z(\vartheta)\Delta_\vartheta = 4\) although \(Z(1) = 2\) and \(Z(\vartheta) = 6\).  Moreover \(\lvert\sigma(\Delta_1)\rvert \le 5\) for every \(\sigma \in \Aut(\C)\), and \(\lvert\Delta_\vartheta\rvert_3 = 3\).
\end{problem}

\section{Heights}\label{sec:heights}

In Section~\ref{sec:general} the archimedean bound of Section~\ref{sec:toy} is used at the trivial twist, and the \(\ell\)-adic bound at a twist that may depend on \(\ell\) and \(\iota\), as in Chapter~\ref{c6-chap}.  The rewritten Lean proof combines the two through the following notion.  Let \(\Ic\) be the set of pairs \((\ell,\iota)\) of a prime \(\ell\) and a field isomorphism \(\iota \colon \C \to \bar\Q_\ell\).  For \(c \in \R\), a family \(y \colon \Ic \to \C\) has \emph{height at most} \(c\) if there are \(H \in \C\), \(\alpha \in \R\) and \(h \colon \N \to \R\) with \(H \ne 0\), \(h(n) \ge 0\) for all \(n\), \(\lvert\sigma H\rvert \le e^{\alpha}\) for every field automorphism \(\sigma\) of \(\C\), \(\lvert\iota(H y_{\ell,\iota})\rvert_\ell \le \ell^{h(\ell)}\) for every \((\ell,\iota) \in \Ic\), and \(\alpha + \sum_{\ell \in T} h(\ell)\log\ell \le c\) for every finite set \(T\) of primes.

\begin{problem}\label{pr:of-mul}
Let \(H \ne 0\) and \(\alpha \in \R\) with \(\lvert\sigma H\rvert \le e^\alpha\) for all \(\sigma\), and let \(g \colon \N \to \R\), of any sign, with \(\lvert\iota(H y_{\ell,\iota})\rvert_\ell \le e^{g(\ell)}\) for all \((\ell,\iota)\) and \(\alpha + \sum_{\ell \in T} g(\ell) \le c\) for every finite set \(T\) of primes.  Then \(y\) has height at most \(c\).
\end{problem}

\begin{problem}\label{pr:height-ops}
If \(y\) and \(y'\) have heights at most \(c\) and \(c'\), respectively, then \(yy'\) has height at most \(c + c'\).  If \(y\) has height at most \(c\), then \(y^n\) has height at most \(nc\) for every \(n \in \N\).  If \(y\) has height at most \(c \le c''\) and \(\lvert\iota(y''_{\ell,\iota})\rvert_\ell \le \lvert\iota(y_{\ell,\iota})\rvert_\ell\) for all \((\ell,\iota)\), then \(y''\) has height at most \(c''\).
\end{problem}

\begin{problem}\label{pr:height-one}
If the constant family \(1\) has height at most \(c\), then \(c \ge 0\).
\end{problem}

\section{The general statement}\label{sec:general}

We take the following theorems from the rewritten Lean proof \cite{GL}, using the notation below; the problems deduce the final height bound.  Let \(P\) and \(D\) be parameters and data as in Chapter~\ref{c32-chap}.  We use from them: the CM field \(E\), its set \(\Phi\) of infinite places and \(f := \#\Phi\); the scales \(x < x_1 < b\), \(L := \log b\), and the natural number \(a = xb\); the weights \(\kappa_\sigma\) at \(\sigma \in \Phi\); the number \(m\) of lifts and the ranks \(r\), \(k := r + 1 - m\) and \(t := 5r\); the integer \(M = p_0p_1\), the level prime \(N\) and the period \(\Omega\); the integer \(q_1\) with \(E = K(\zeta_{q_1})\), where \(K = \Q(\sqrt{-d})\).  Put
\[
  s_\infty(n) := \log n! + n\log(M/\Omega), \qquad
  \sfin(y,\sigma) := \sum_{\nu=1}^{a} \Lambda(\nu)\bigl(1 - \chi_K(\nu)\bigr)\Bigl(\frac{y}{\nu} - \kappa_{-\nu\sigma}\Bigr)_{\!+},
\]
where \(-\nu\sigma\) is the product of \(\sigma\) with the infinite place attached to \(-\nu\) modulo \(q_1\) (the identity place if \(\nu\) is not prime to \(q_1\)), and
\[
  \hdir := \sum_{\sigma \in \Phi} \bigl(s_\infty(\kappa_\sigma) - \sfin(\kappa_\sigma,\sigma)\bigr), \qquad
  B := \frac{faL^3}{mk} .
\]
Here \(s_\infty(n)\) is the size of a derivative of order \(n\) and \(\sfin\) the prime sum that measures its guaranteed divisibility; \(\hdir\) is the \emph{direction amount} and \(B\) the \emph{denominator budget}.

\paragraph{Twists and lifts.}  A \emph{tame character} \(\vartheta\) consists of an automorphic character of the norm-one torus \(U(1)\) for \(E/E^+\), with \(E^+\) the maximal totally real subfield of \(E\), of weight zero, a finite set \(Q_\vartheta\) of finite places with \(\bar w \notin Q_\vartheta\) for \(w \in Q_\vartheta\) and \(N_{Q_\vartheta} := \prod_{w \in Q_\vartheta} q_w\) prime to \(MN\), and residue characters \(\psi_w\), trivial for \(w \notin Q_\vartheta\), that give its tame part.  It is a \emph{twist} if \(\psi_w \ne 1\) for \(w \in Q_\vartheta\), and an \emph{\(m\)-fold twist} if \(\psi_w^n \ne 1\) for \(w \in Q_\vartheta\) and \(1 \le n \le m\); then \(\xi^j\) is a twist for \(1 \le j \le m\).  The trivial twist \(1\) has \(Q_1 = \varnothing\).  For \(1 \le j \le m\) and a twist \(\vartheta\), let \(\Theta_{\vartheta j} \colon G_j \to A_{\vartheta j}\) be the lift of Chapter~\ref{c37-chap} at the twist \(\vartheta\), from the good forms on the Hermitian space \(W_j\) of rank \(r\) to the algebraic forms on the definite space of rank \(r + t\) with the weight, level and character of its datum \(O_{\vartheta j}\).  It is injective, and \(\dim G_j = d_T\), the dimension of the good target forms.  Let \(R_{\vartheta j}\) be its image, \(\pr_{\vartheta j}\) the orthogonal projection onto it, and \(Z_{\vartheta j} := \det(\Theta_{\vartheta j}^*\Theta_{\vartheta j})\).  For an \(m\)-fold twist \(\xi\), \(Z_j(\xi) := Z_{\xi^j j}\), as in Chapter~\ref{c37-chap}; in particular \(Z_j(1) = Z_{1j}\).

For a ring homomorphism \(\eta \colon \C \to \C_\ell\), a form \(y \in A_{\vartheta j}\) is \emph{integral at} \(\eta\) if, for every finite adelic point \(g\) of the definite unitary group, the vector \(\eta(y(g))\) moved by \(g^{-1}\) in the weight representation has coordinates of absolute value at most \(1\).  For \((\ell,\iota) \in \Ic\), integral at \(\iota\) means integral at the composite of \(\iota\) with the inclusion \(\bar\Q_\ell \subset \C_\ell\).  For \(\sigma \in \Aut(\C)\), \(N_\sigma(y)\) is the squared norm of \(y\) computed after applying \(\sigma\) to its values: a weighted sum over classes of the integral Weyl Gram form.  Let \(w\) be a finite place of \(E\) with \(\bar w \ne w\), not among the places of \(O_{\vartheta j}\) or their conjugates.  The \emph{false lifts at} \(w\) are the forms in the image of the idempotent \(e_0\) that are killed by the Hecke polynomial \(H_w\) of \(O_{\vartheta j}\) at its test value \(\lambda_w\), and are orthogonal to \(R_{\vartheta j}\); the lifts lie in \(\im e_0 \cap \ker H_w(\lambda_w)\), which plays the part of \(X\) in Section~\ref{sec:toy}.  A form is \emph{orthogonal to the false lifts} if it is orthogonal to the false lifts at every such \(w\).

\paragraph{Hecke values.}  For \(1 \le j \le m\) and a twist \(\vartheta\), let \(S := S_M \cup Q_\vartheta\), where \(S_M\) is the set of places of \(E\) dividing the fixed element \(\pi_M\) of norm \(M\) from \(K\), and \(q_w\) is the cardinality of the residue field at \(w\).  Put
\[
  H_j(\vartheta) := \prod_{i \in \{\varnothing, +, -\}} \biggl[\Bigl(\prod_{w \in S} q_w\Bigr)^{z_i^2} \prod_{n=1}^{z_i} \bigl\lvert L^{S}\bigl(n, (\nu_i\vartheta^{\epsilon_i})^{-1}\bigr)\bigr\rvert^2\biggr] ,
\]
where \(\epsilon_\varnothing = \epsilon_+ = 1\), \(\epsilon_- = -1\), \(z_\varnothing = 7\lfloor m/2\rfloor + \lfloor k/2\rfloor\), \(z_+ = m\), \(z_- = \lfloor m/2\rfloor\), \(\nu_\varnothing := \mu_j^0\upsilon^{j + \lfloor(t - 11m - 1)/2\rfloor}\) and \(\nu_\pm := \upsilon^{-z_\pm}\chi_M^{\pm j}\).  Here \(\mu_j^0\) is the untwisted character of the \(j\)-th lift, \(\upsilon\) the restriction to \(U(1)\) of the splitting character of the Weil data, and \(\chi_M\) the fixed character of weight zero whose residue characters \(\omega_w\) are trivial off \(S_M\); for the characters \(\mu\) used here, the Hecke character \(x \mapsto \mu(x/\bar x)\) of \(E\) is unramified outside \(S\) and the conjugates of its places, and \(L^S(n,\mu)\) is its \(L\)-function without the Euler factors at these places.  Put \(\rho_\vartheta := H_j(\vartheta)/H_j(1)\).

\paragraph{Comparison operators.}  Fix \(1 \le j \le m\).  A family of operators \(U_\vartheta \in \End_\C A_{\vartheta j}\), one for every twist \(\vartheta\), is a \emph{comparison operator} if:
(1) \(\rho_\vartheta \ne 0\) for every twist \(\vartheta\);
(2) some \(U_0 \in \End G_j\) has \(\langle y, U_0 y\rangle > 0\) for \(y \ne 0\) and \(\langle\Theta_{\vartheta j}y', U_\vartheta\Theta_{\vartheta j}y\rangle = \rho_\vartheta\langle y', U_0 y\rangle\) for every twist \(\vartheta\) and all \(y', y \in G_j\);
(3) for every twist \(\vartheta\) and \(z \in R_{\vartheta j}\), \(U_\vartheta z\) is orthogonal to the false lifts;
(4) for every twist \(\vartheta\), every prime \(\ell\) prime to \(N_{Q_\vartheta}\), and every \(\eta \colon \C \to \C_\ell\), \(U_\vartheta\) maps forms integral at \(\eta\) to forms integral at \(\eta\);
(5) \(N_\sigma(U_1\varphi) \le \exp(3fmbL^2)^2 N_\sigma(\varphi)\) for every \(\varphi \in A_{1j}\) and every \(\sigma \in \Aut(\C)\).

\begin{theorem}\label{thm:operator}
For every \(1 \le j \le m\) the operators \(U_\vartheta := n\, h^2\, g_\vartheta g_\vartheta^*\) form a comparison operator.  Here \(g_\vartheta\) is a gluing map from tuples of forms on a smaller definite space to \(A_{\vartheta j}\), \(h\) is the hook number of the diagram of the weight of those forms, and \(n := \prod_{w \in S_M} \#\mathrm{GL}_{rt}(\mathcal O_E/w)\), a common multiple of the orders of the stabilizers at the levels used.
\end{theorem}

The rewriting proves (1) and (2) by the mechanism of Problem~\ref{pr:rank-one}, with \(g_\vartheta^*\Theta_{\vartheta j} = c_\vartheta^{\prime\,-1}\Theta^\circ\) for a nonzero scalar \(c'_\vartheta\) and an injective map \(\Theta^\circ\) independent of \(\vartheta\); and (4) and (5) by writing \(U_\vartheta\) as a sum of operators of rank at most one over a tight frame of tuples.

For a comparison operator put \(\Delta_\vartheta := \det(\pr_{\vartheta j}U_\vartheta|_{R_{\vartheta j}})\).  For \((\ell,\iota) \in \Ic\), a twist \(\vartheta\) and \(\beta \in \R\), say that \(\beta\) \emph{clears the projection} at \((\iota,\vartheta,j)\) if some \(\Lambda \in \End R_{\vartheta j}\) has \(\lvert\iota(\det\Lambda)\rvert_\ell \ge e^{-(\dim R_{\vartheta j})\,r\beta}\) and \(\Lambda(\pr_{\vartheta j}y)\) is integral at \(\iota\) for every \(y\) that is integral at \(\iota\) and orthogonal to the false lifts.

\begin{theorem}\label{thm:compression}
Let \(1 \le j \le m\) and let \((U_\vartheta)\) be a comparison operator.
\textup{(a)} There is \(C \in \C\), \(C \ne 0\), with \(Z_{\vartheta j}\Delta_\vartheta = C\rho_\vartheta^{d_T}\) for every twist \(\vartheta\).
\textup{(b)} For every twist \(\vartheta\), every \((\ell,\iota) \in \Ic\) with \(\ell\) prime to \(N_{Q_\vartheta}\), and every \(\beta\) that clears the projection at \((\iota,\vartheta,j)\), \(\lvert\iota(\Delta_\vartheta)\rvert_\ell \le e^{d_T r\beta}\).
\textup{(c)} \(\lvert\sigma(\Delta_1)\rvert \le e^{3d_T fmbL^2}\) for every \(\sigma \in \Aut(\C)\).
\end{theorem}

The rewriting proves (a)--(c) along the lines of Section~\ref{sec:compress} and Section~\ref{sec:toy}; for (c) it uses that \(N_\sigma(\pr_{\vartheta j}y) \le N_\sigma(y)\) for every \(y\) orthogonal to the false lifts.

\paragraph{Ratios of \(L\)-values.}  A \emph{critical pair} \(p\) consists of an automorphic character \(\nu\) of the norm-one torus \(U(1)\) of weight \(-c\), with \(c \colon \Phi \to \N\), residue characters \(\psi_w\) that give its tame part and are trivial off \(S_M\), and an integer \(z\) with \(1 \le z \le t\) and \(z \le c_\sigma \le 4b\) for all \(\sigma\).  It is \emph{ramified} if \(\psi_w \ne 1\) exactly for \(w \in S_M\), and \emph{main for} \(j\) if \(z \ge 10\), \(m < z \le r\), \(1 \le j \le m\), \(\psi = \omega^j\) for the fixed residue characters \(\omega\) of \(\chi_M\), and \(w_j(\sigma) + r - t \le c_\sigma \le w_j(\sigma)\) for all \(\sigma\), where \(-w_j\) is the weight of the character of the \(j\)-th lift.  For a tame character \(\vartheta\) and \(s = \pm1\) put
\[
  R_p(\vartheta, s) := \frac{\lvert L^{S_M \cup Q_\vartheta}(z, (\nu\vartheta^{s})^{-1})\rvert}{\lvert L^{S_M}(z, \nu^{-1})\rvert}, \qquad R_p(\vartheta) := R_p(\vartheta, 1) ,
\]
the Euler factors at the conjugate places being omitted as well.  For a critical pair with \(z \ge 2\), let \(\phi_\vartheta\) be the lift of the constant form \(1\) on \(U(1)\) to the definite space of rank \(2z\), for the datum with character \(\nu\vartheta\), multiplied by the index of the chosen level and by a fixed scale of the pair, both independent of \(\vartheta\).  The rewritten Lean proof shows that \(R_p(\vartheta) = (\lVert\phi_\vartheta\rVert^2/N_{Q_\vartheta}^{2z-1})/\lVert\phi_1\rVert^2\) for every tame \(\vartheta\); so \(R_p(\vartheta)^{-1} = N_{Q_\vartheta}^{2z-1}\lVert\phi_1\rVert^2/\lVert\phi_\vartheta\rVert^2\), and the inverse ratios are, up to a power of \(N_{Q_\vartheta}\), ratios of squared norms of lifts.

\begin{theorem}\label{thm:factor}
For every \(1 \le j \le m\) there are \(e \in \N\), finite subsets \(A_{\mathrm{main}} \subset \N\) and \(A_{\mathrm{ram}} \subset \{\varnothing,+,-\} \times \N\), families of critical pairs \((p_a)_{a \in \N}\) and \((q_b)_{b \in \{\varnothing,+,-\} \times \N}\), and signs \((s_b)_{b \in \{\varnothing,+,-\} \times \N}\), with \(2\#A_{\mathrm{main}} = k\) and \(\#A_{\mathrm{ram}} = 5m\), such that \(p_a\) is main for \(j\) for every \(a \in A_{\mathrm{main}}\), \(q_b\) is ramified for every \(b\), and for every twist \(\vartheta\)
\[
  H_j(\vartheta) = N_{Q_\vartheta}^{e}\Bigl(\prod_{a \in A_{\mathrm{main}}} R_{p_a}(\vartheta)\prod_{b \in A_{\mathrm{ram}}} R_{q_b}(\vartheta, s_b)\Bigr)^2 H_j(1) .
\]
\end{theorem}

\paragraph{Regular families.}  Let \(\theta\) assign an \(m\)-fold twist \(\theta_{\ell,\iota}\) to every \((\ell,\iota) \in \Ic\).  It is \emph{regular} if:
(i) \(\ell\) is prime to \(N_{Q_{\theta_{\ell,\iota}}}\) for every \((\ell,\iota)\);
(ii) for every \(1 \le j \le m\), every ramified critical pair \(q\) and \(s = \pm1\), the family \((\ell,\iota) \mapsto R_q(\theta_{\ell,\iota}^j, s)^{-1}\) has height at most \(x_1\);
(iii) for every \(1 \le j \le m\) there is \(\gamma \colon \N \to \R\) with \(\sum_{\ell \in T}\gamma(\ell) \le B\) for every finite set \(T\) of primes, such that for every \((\ell,\iota)\), with \(\vartheta := \theta_{\ell,\iota}^j\): \(\gamma(\ell)\) clears the projection at \((\iota,\vartheta,j)\); and for every critical pair \(p\) main for \(j\) and any two test forms \(\Xi, \Xi'\) for \(\ell\) in the space of the lift of \(\nu\vartheta\) to the definite space of rank \(2z\), \(\lvert\iota\langle\pr\Xi, \pr\Xi'\rangle\rvert_\ell \le e^{\gamma(\ell)}\), where \(\pr\) is the orthogonal projection onto the line of that lift.
Here, for the datum \(O\) of that lift, a \emph{test form} for \(\ell\) is a form that is integral at every ring homomorphism \(\eta \colon \C \to \C_\ell\) and, at every place \(w\) with \(\bar w \ne w\) off the places of \(O\) and their conjugates, orthogonal to \(\im e_{X-1} \cap \ker H_w(\lambda_w) \cap \ker O^*\); here \(e_{X-1}\) is the idempotent with label \(X - 1\) at every chosen place (in rank one, the mean over the product of the multiplicative groups of the residue fields), and \(O^*\) is the adjoint of the lift.  Chapter~\ref{c39-chap} treats the denominators in (iii), and Chapter~\ref{c40-chap} the existence of regular families.

\begin{theorem}\label{thm:main-pair}
Let \(\theta\) be regular, \(1 \le j \le m\), and \(p\) a critical pair main for \(j\).  Then the family \((\ell,\iota) \mapsto R_p(\theta_{\ell,\iota}^j)^{-1}\) has height at most \(6B - \hdir\).
\end{theorem}

The rewriting uses (i) and the coefficient estimate of Theorem~\ref{c18-thm:attained} to obtain the test forms.  Up to a power of \(N_{Q_{\theta_{\ell,\iota}^j}}\), it expresses \(R_p(\theta_{\ell,\iota}^j)^{-1}\) as a product of four families.  Conditions (ii) and (iii) enter the estimates for these factors, whose height bounds sum to at most \(6B - \hdir\).

\paragraph{The cost of one lift.}  Let \(\theta\) be regular and \(1 \le j \le m\), let \((U_\vartheta)\) be the comparison operator of Theorem~\ref{thm:operator}, and write \(\vartheta_{\ell,\iota} := \theta_{\ell,\iota}^j\).

\begin{problem}\label{pr:delta-height}
The family \((\ell,\iota) \mapsto \Delta_{\vartheta_{\ell,\iota}}/\Delta_1\) has height at most \(d_T(3fmbL^2 + rB)\).
\end{problem}

\begin{problem}\label{pr:hecke-height}
The family \((\ell,\iota) \mapsto \rho_{\vartheta_{\ell,\iota}}^{-1}\) has height at most \(10mx_1 + k(6B - \hdir)\).
\end{problem}

\begin{problem}\label{pr:identity}
For every \(m\)-fold twist \(\xi\), \(Z_j(1)/Z_j(\xi) = (\Delta_{\xi^j}/\Delta_1)\,\rho_{\xi^j}^{-d_T}\).
\end{problem}

It remains to compare the constants.  The following inequality is derived here from the sizes of the data.

\begin{problem}\label{pr:budget}
Let \(f \ge 1\), \(b > 1\) with \(L = \log b \ge 16\), \(x \ge 1\), \(\sqrt x \le m \le 2\sqrt x\), \(r \ge 2m\), \(k = r + 1 - m\), \(1 \le x_1 \le b\), \(a = xb\) and \(B = faL^3/(mk)\).  Then for every real \(\hdir\)
\[
  3fmbL^2 + rB + 10mx_1 + k(6B - \hdir) \le k(LB - \hdir) .
\]
The data satisfy these hypotheses.
\end{problem}

Problems~\ref{pr:delta-height}, \ref{pr:hecke-height}, \ref{pr:identity}, \ref{pr:height-ops} and~\ref{pr:budget} give the following height bound.

\begin{theorem}\label{thm:main}
For all parameters \(P\) and data \(D\), every regular family \(\theta\) of \(m\)-fold twists, and every \(1 \le j \le m\), the family
\[
  (\ell,\iota) \longmapsto \frac{Z_j(1)}{Z_j(\theta_{\ell,\iota})}
\]
has height at most \(d_T\, k\,(LB - \hdir)\).
\end{theorem}

\section*{Hints}

\subsection*{\S\ref{sec:compress}}
\begin{description}
\item[\ref{pr:compression}] Since \(u^*\) kills \(R^\perp\), \(u^*Tu = u^*\pr_R T u\).  Write \(u\) as an isomorphism \(A \to R\) followed by the inclusion.
\item[\ref{pr:comparison}] Problem~\ref{pr:compression} with \(T = U_\vartheta\); the hypothesis says \(u_\vartheta^*U_\vartheta u_\vartheta = \rho_\vartheta U_0\).
\item[\ref{pr:rank-one}] \(g_\vartheta^*v = \langle w_\vartheta, v\rangle\).
\item[\ref{pr:eigen}] Factor the characteristic polynomial over \(\C\).
\end{description}

\subsection*{\S\ref{sec:toy}}
\begin{description}
\item[\ref{pr:false}] \(X = R \oplus F\), orthogonally.
\item[\ref{pr:conj}] \(\pr_X = M(M^*M)^{-1}M^*\) for a matrix \(M\) whose columns form a basis of \(X\) and have coordinates in \(K\); on \(K\), complex conjugation commutes with \(\sigma\).
\item[\ref{pr:arch}] Take an eigenvector \(z\) of \(\pr_R U|_R\) and apply Problems~\ref{pr:false} and~\ref{pr:conj} to \(Uz\); then Problem~\ref{pr:eigen}.
\item[\ref{pr:padic}] Scale an eigenvector of \(\Lambda\pr_R U|_R\) to be integral at \(\iota\) with a coordinate of absolute value \(1\).
\item[\ref{pr:clear}] Every vector of \(\Lc_\iota\) is a combination of \(e_1, \dots, e_n\) with coefficients \(t_i\) satisfying \(\lvert\iota t_i\rvert_\ell \le 1\); use Problem~\ref{pr:false}.
\item[\ref{pr:toy-cong}] Problem~\ref{c28-pr:symmetry}: the index of \((\Z^3 \cap X) \oplus \Z(1,1,1)\) in \(\Z^3\).
\item[\ref{pr:toy-delta}] The false lifts for \(R_1\) are spanned by \(v_\vartheta\), and conversely; use Problem~\ref{pr:rank-one}.  For the bounds, \(\lVert w_1\rVert^2 = 5\), and Problem~\ref{pr:toy-cong}.
\end{description}

\subsection*{\S\ref{sec:heights}}
\begin{description}
\item[\ref{pr:of-mul}] Try \(h(\ell) := \max(g(\ell), 0)/\log\ell\) for primes \(\ell\), and \(h(n) := 0\) for the other \(n\).
\item[\ref{pr:height-ops}] Multiply the multipliers \(H\).
\item[\ref{pr:height-one}] \(H\) is algebraic, since \(\Aut(\C)\) moves a transcendental number to any other; apply the product formula to \(H\) in a Galois number field, noting that every embedding of that field into \(\bar\Q_\ell\) extends to some \(\iota\).
\end{description}

\subsection*{\S\ref{sec:general}}
\begin{description}
\item[\ref{pr:delta-height}] Problem~\ref{pr:of-mul} with \(H = \Delta_1\), Theorem~\ref{thm:compression}(b) and~(c), and the function \(\gamma\) of~(iii).

\end{description}
