\medskip
How does the determinant \(Z(\vartheta)\) of the chain change with the \(m\)-fold twist \(\vartheta\)?  By Theorem~\ref{thm:main}, its quotient by \(Z(1)\) is the product of the \(m\) quotients \(Z_j(\vartheta)/Z_j(1)\) for single lifts, times a correction factor: a product of pairwise ratios \(\rho_{ij}(\vartheta)\) of Hecke \(L\)-values, multiplied by \((\Nm Q_\vartheta)^{e_Z/2}\) with an integer \(e_Z\) independent of \(\vartheta\), and raised to the power \(d_T\).  The problems derive Theorem~\ref{thm:main} from three statements of the following chapters: the Gram identity of one lift (Theorem~\ref{c1-thm:one-lift}), the comparison of the good forms on \(W_j\) and on \(Y_0\) (Corollary~\ref{c2-cor:equal}), and the Gram identity of the chain (Theorem~\ref{c4-thm:chain-gram}), which rests on the exchange identity (Theorem~\ref{c3-thm:exchange}).  The proof also states Theorem~\ref{thm:main}.  Section~\ref{sec:volumes} treats the linear algebra in its simplest setting; Sections~\ref{sec:one-lift}--\ref{sec:factor} assemble the theorem.  Section~\ref{sec:aux} gives two further results used in the proof of Theorem~C; both are read from the proof (Theorems~\ref{thm:aux} and~\ref{thm:small-ratios}).

\section{Volumes of maps}\label{sec:volumes}

For a linear map \(u \colon U \to V\) of finite-dimensional complex inner product spaces let
\[
  Z(u) := \det\nolimits_U (u^* u) ,
\]
the determinant on the domain \(U\); we call it the \emph{volume} of \(u\).  Dimensions are complex dimensions, the determinant on the zero space is \(1\), and inner products are conjugate-linear in the first argument and linear in the second.

\begin{problem}\label{pr:gram}
For every orthonormal basis \(e_1, \dots, e_d\) of \(U\), \(Z(u) = \det\bigl(\langle u e_a, u e_b \rangle\bigr)_{a,b}\).  The number \(Z(u)\) is real and \(\ge 0\), and \(Z(u) > 0\) if and only if \(u\) is injective.
\end{problem}

\begin{problem}\label{pr:scalar-gram}
Let \(S\) be an endomorphism of \(U\) and \(a \in \C\) with \(\langle ux, uy\rangle = a \langle x, Sy\rangle\) for all \(x, y \in U\).  Then \(Z(u) = a^{\dim U} \det S\).
\end{problem}

\begin{problem}\label{pr:square-chain}
If \(\theta_j \colon V_{j-1} \to V_j\), for \(1 \le j \le m\), are linear maps between spaces of one and the same dimension, then \(Z(\theta_m \circ \dots \circ \theta_1) = \prod_{j=1}^m Z(\theta_j)\).
\end{problem}

When the dimension grows along the chain, equality can fail.

\begin{problem}\label{pr:position}
Let \(\theta_2 \colon \C^2 \to \C^2\) be \(\operatorname{diag}(1, 2)\), and let \(\theta_1, \theta_1' \colon \C \to \C^2\) be \(x \mapsto (x, 0)\) and \(x \mapsto (0, x)\).  Then \(Z(\theta_1) = Z(\theta_1') = 1\) and \(Z(\theta_2) = 4\), while \(Z(\theta_2\theta_1) = 1\) and \(Z(\theta_2\theta_1') = 4\).
\end{problem}

What replaces Problem~\ref{pr:square-chain} is an identity for \(\Theta^*\Theta\) on the first space itself (Theorem~\ref{c4-thm:chain-gram}); Problem~\ref{c4-pr:push-down} is a model of how such an identity can be propagated along a chain.

For \(a \ge 0\) divisible by \(4\), the two lifts from \(\mathrm U(1)\) of Chapter~\ref{c22-chap} used here have a one-dimensional source.  The lift \(\Theta_{(a)}\) from \(\mathrm U(1)\) to \(\mathrm U(1)\) is multiplication by \(c_a/4\) (Problem~\ref{c22-pr:U1U1}), and by Problem~\ref{c22-pr:rallis} the lift \(\Theta_{(a)}\) from \(\mathrm U(1)\) to \(\mathrm U(2)\) satisfies
\[
  \lVert \Theta_{(a)}\varphi \rVert^2 = \lambda_a \lVert \varphi \rVert^2 , \qquad
  \lambda_a := \frac{1}{128} \sum_j \binom{a}{j} c_j^2 c_{a-j}^2 .
\]
This finite calculation illustrates the inner product of a lift with itself; the strict rank inequality of the inner product formula (Theorem~\ref{c1-thm:rallis}) first allows a source of rank \(1\) when the target has rank \(3\), the case with which Chapter~\ref{c1-chap} begins.

\begin{problem}\label{pr:smallest}
For \(4 \mid a\), the composite of the lift \(\mathrm U(1) \to \mathrm U(1)\) with the lift \(\mathrm U(1) \to \mathrm U(2)\) of Chapter~\ref{c22-chap}, in degree \(a\), has volume \(\lambda_a c_a^2/16\), the product of the two volumes.
\end{problem}

\section{One lift}\label{sec:one-lift}

For \(1 \le j \le m\) and a tame character \(\xi\), the single lift \(\hat u_{\xi,j}\) on \(H(W_j)\) and its restriction \(u_{\xi,j}\) to \(H_g(W_j)\) are those of Section~\ref{ob-sec:chain}; \(H(V)\) carries the Petersson product (Section~\ref{c2-sec:traces}).  We recall two statements.  By Theorem~\ref{c1-thm:one-lift}, for every \(1 \le j \le m\) there is \(a_j \in \C\), \(a_j \ne 0\), such that for every tame character \(\xi\),
\[
  \hat u_{\xi,j}^*\, \hat u_{\xi,j} = a_j\, c_{Q_\xi}\, \Rc_j(\xi; W_j)
\]
on \(H(W_j)\), with \(c_Q\) the twist factor of Definition~\ref{c1-def:twist-factor} and \(\Rc_j(\xi; V)\) the Rallis operator of Definition~\ref{c1-def:rallis-operator}; and \(\hat u_{\xi,j}\) is injective.  By Corollary~\ref{c2-cor:equal}, for \(1 \le j \le m\): \(\dim H_g(W_j) = d_T\), and for every tame character \(\xi\), \(\Rc_j(\xi; Y_0)\) maps \(T\) to itself, \(\Rc_j(\xi; W_j)\) maps \(H_g(W_j)\) to itself, and
\[
  \det\bigl(\Rc_j(\xi; W_j) \mid H_g(W_j)\bigr) = \det\bigl(\Rc_j(\xi; Y_0) \mid T\bigr) .
\]

\begin{problem}\label{pr:one-lift}
For \(1 \le j \le m\) and every tame character \(\xi\), \(Z(u_{\xi,j}) = (a_j c_{Q_\xi})^{d_T} \det\bigl(\Rc_j(\xi; Y_0) \mid T\bigr)\).  Moreover \(Z(u_{\xi,j}) > 0\).
\end{problem}

\section{Twists and the ratios}\label{sec:twists}

Tame characters \(\xi\), their sets \(Q_\xi\) and powers \(\xi^n\), twists and \(m\)-fold twists \(\vartheta\) are as in Chapter~\ref{ob-chap}; the power \(\xi^n\) has the same set \(Q_\xi\).

\begin{problem}\label{pr:powers}
If \(\vartheta\) is an \(m\)-fold twist and \(1 \le n \le m\), then \(\vartheta^n\) is a twist with the same set \(Q_\vartheta\), so \(c_{Q_{\vartheta^n}} = c_{Q_\vartheta}\).
\end{problem}

For \(1 \le i < j \le m\) and an \(m\)-fold twist \(\vartheta\), \(\rho_{ij}(\vartheta)\) is the ratio of Definition~\ref{c4-def:rho}: with \(\nu := \mu_j^0/\mu_i^0\),
\[
  \rho_{ij}(\vartheta) := \prod_{z=1}^{t} \frac{\lvert L^{\mathcal M \cup Q_\vartheta}\bigl(z, (\nu\vartheta^{j-i})^{-1}\bigr)\rvert}{\lvert L^{\mathcal M}(z, \nu^{-1})\rvert} ,
\]
a nonnegative real number.  These ratios enter the Gram identity of the chain through the exchange identity (Theorem~\ref{c3-thm:exchange}), whose scalar has squared modulus a constant times the exchange factor \(\mathrm{ex}_h(S,\nu)\) of Definition~\ref{c3-def:exchange-factor}; Chapter~\ref{c4-chap} applies it with \(h = t\) and \(S = \mathcal M \cup Q_\vartheta\), once for each pair \(i < j\).

\begin{problem}\label{pr:rho-one}
\(\rho_{ij}(1) = 1\) for \(1 \le i < j \le m\).
\end{problem}

\section{The factorization}\label{sec:factor}

For an \(m\)-fold twist \(\vartheta\), the lower chain \(\Theta_\vartheta\) of Section~\ref{ob-sec:chain}, restricted to \(T\), takes values in a space of algebraic level forms on \(Y_m\) with the class-sum inner product; it is injective (Problem~\ref{c4-pr:theta-injective}).  In the notation of Section~\ref{sec:volumes}, \(Z(\vartheta) = Z(\Theta_\vartheta|_T)\) and \(Z_j(\vartheta) = Z(u_{\vartheta^j, j})\).  Let \(\Rc_\vartheta = \Rc_1(\vartheta; Y_0)\,\Rc_2(\vartheta^2; Y_0) \cdots \Rc_m(\vartheta^m; Y_0)\) be the endomorphism of \(T\) of Definition~\ref{c4-def:rallis-product}.  By Theorem~\ref{c4-thm:chain-gram} there is a function \(c\) from \(m\)-fold twists to \(\C\) with \(c \doteq \prod_{1 \le i < j \le m}\rho_{ij}\), that is (Definition~\ref{c4-def:doteq}), with a real number \(C'\) and an integer \(e'\) such that \(\lvert c(\vartheta)\rvert^2 = C'(\Nm Q_\vartheta)^{e'}\bigl(\prod_{1 \le i < j \le m}\rho_{ij}(\vartheta)\bigr)^2\) for every \(m\)-fold twist \(\vartheta\), and such that for every \(m\)-fold twist \(\vartheta\) and all \(x, y \in T\),
\[
  \langle \Theta_\vartheta x, \Theta_\vartheta y\rangle = c(\vartheta)\,c_{Q_\vartheta}^m\,\langle x, \Rc_\vartheta y\rangle .
\]

\begin{problem}\label{pr:abs-ratio}
Let \(X\) be a set with a point \(1\), let \(N \colon X \to \R_{>0}\) and \(f \colon X \to \R_{\ge 0}\) with \(N(1) = f(1) = 1\), and let \(c \colon X \to \C\), \(C' \in \R\) and \(e' \in \Z\) satisfy \(\lvert c(x) \rvert^2 = C' N(x)^{e'} f(x)^2\) for all \(x\).  If \(c(1) \ne 0\), then \(C' = \lvert c(1)\rvert^2\) and \(\lvert c(x)/c(1) \rvert = N(x)^{e'/2} f(x)\) for all \(x\).
\end{problem}

\begin{problem}\label{pr:chain-det}
In Theorem~\ref{c4-thm:chain-gram}, if \(c(1) = 0\) then \(T = 0\).  For every \(m\)-fold twist \(\vartheta\),
\[
  Z(\vartheta) = \Bigl( c(\vartheta) \Big/ \prod_{j=1}^m a_j \Bigr)^{d_T} \prod_{j=1}^m Z_j(\vartheta) ,
\]
and \(Z(\vartheta) > 0\).
\end{problem}

\begin{problem}\label{pr:quotient}
If \(T \ne 0\), then for every \(m\)-fold twist \(\vartheta\),
\[
  \frac{Z(\vartheta)}{Z(1)} = \Bigl( (\Nm Q_\vartheta)^{e'/2} \prod_{1 \le i < j \le m} \rho_{ij}(\vartheta) \Bigr)^{d_T} \prod_{j=1}^m \frac{Z_j(\vartheta)}{Z_j(1)} .
\]
If \(T = 0\), both sides equal \(1\) (a real number to the power \(0\) is \(1\)).
\end{problem}

The proof states the normalized determinant identity as follows.

\begin{theorem}\label{thm:main}
Let \(P\) be parameters and \(D\) admissible data for \(P\).  There is an integer \(e_Z\) such that for every \(m\)-fold twist \(\vartheta\):
\begin{enumerate}
\item[(a)] \(0 < Z(\vartheta)\);
\item[(b)] \(0 < Z_j(\vartheta)\) for every integer \(j\) with \(1 \le j \le m\);
\item[(c)] in \(\C\),
\[
  \frac{Z(\vartheta)}{Z(1)} = \biggl( (\Nm Q_\vartheta)^{e_Z/2} \prod_{i=1}^{m} \prod_{j=i+1}^{m} \rho_{ij}(\vartheta) \biggr)^{d_T} \prod_{j=1}^m \frac{Z_j(\vartheta)}{Z_j(1)} .
\]
\end{enumerate}
\end{theorem}

Here \(Z(\vartheta)\) and \(Z_j(\vartheta)\) are complex numbers by definition, and (a), (b) say that they are real and positive; the base of the power \(d_T\) in (c) is a real number, with \(e_Z/2\) a real exponent, and \(e_Z\) does not depend on \(\vartheta\).  Problems~\ref{pr:one-lift} and~\ref{pr:abs-ratio}--\ref{pr:quotient}, with Theorem~\ref{c1-thm:one-lift}, Corollary~\ref{c2-cor:equal} and Theorem~\ref{c4-thm:chain-gram}, prove Theorem~\ref{thm:main}; one may take \(e_Z = e'\) when \(T \ne 0\).  Thus the volume of the chain at a twist, relative to the trivial twist, is expressed through \(m\) volumes of single lifts, a power of \(\Nm Q_\vartheta\), and \(m(m-1)/2\) ratios \(\rho_{ij}\) of \(L\)-values.

\section{Auxiliary forms}\label{sec:aux}

The proof of Theorem~C extends a linearly independent family of target forms to a basis.  For each \(m\)-fold twist \(\vartheta\), a common nonzero scalar makes the \(\vartheta\)-lifts of the added forms integral at every \(\iota\mid\ell\) with \(\ell\nmid\Nm Q_\vartheta\).  Both statements of this section are read from the proof.  The \emph{turn ratios} are
\[
  \rho'_{ij}(\vartheta) := \prod_{z=1}^{t} \frac{\lvert L^{\mathcal M \cup Q_\vartheta}\bigl(z, (\mu_{i+1}^0\mu_{j+1}^0\,\vartheta^{j+1})^{-1}\bigr)\rvert}{\lvert L^{\mathcal M}\bigl(z, (\mu_{i+1}^0\mu_{j+1}^0)^{-1}\bigr)\rvert}, \qquad 0 \le i, j < m ,
\]
a factor with zero denominator being \(0\).
For an automorphism \(\sigma\) of \(\C\) write \(\lVert y \rVert_\sigma^2\) for the squared norm of the conjugate of a form \(y\) by \(\sigma\).

\begin{theorem}[auxiliary forms]\label{thm:aux}
There are an integer \(e_\gamma\) and numbers \(\gamma(\vartheta) \in \C^\times\), one for each \(m\)-fold twist, with \(\gamma(1) = 1\) and
\[
  \lvert\gamma(\vartheta)\rvert = (\Nm Q_\vartheta)^{e_\gamma/2} \prod_{0 \le i, j < m} \rho'_{ij}(\vartheta) ,
\]
such that \(T\) is spanned by the forms \(y\) with the two properties: \(\gamma(\vartheta)\Theta_\vartheta y\) is integral at every \(\iota \mid \ell\) with \(\ell \nmid \Nm Q_\vartheta\), for every \(m\)-fold twist \(\vartheta\); and \(\lVert\Theta_1 y\rVert_\sigma^2 \le \exp(2x_1^2b^2)\) for every automorphism \(\sigma\) of \(\C\).
\end{theorem}

\begin{theorem}[small ratios]\label{thm:small-ratios}
Let \(\theta\) be a family of \(m\)-fold twists with small ratios (Definition~\ref{c38-def:small-ratios}).  For every integer \(e\), the families
\[
  \iota \mapsto \Bigl((\Nm Q_{\theta_\iota})^{e/2}\prod_{1 \le i < j \le m}\rho_{ij}(\theta_\iota)\Bigr)^{-1}
  \quad\text{and}\quad
  \iota \mapsto \Bigl((\Nm Q_{\theta_\iota})^{e/2}\prod_{0 \le i, j < m}\rho'_{ij}(\theta_\iota)\Bigr)^{-1}
\]
have height at most \(m^2tx_1\).
\end{theorem}

By Problem~\ref{c4-pr:rho-pair} each \(\rho_{ij}\) is a product of \(t\) ratios \(R_P(\vartheta^{j-i}, 1)\) with \(P\) a ramified critical pair; the proof shows that each \(\rho'_{ij}\) is likewise a product of \(t\) ratios \(R_P(\vartheta^n, 1)\), with \(1 \le n \le m\) and \(P\) a ramified critical pair.

\begin{problem}\label{pr:count-ratios}
Let \(\theta\) be a family of \(m\)-fold twists such that, for every \(1 \le n \le m\) and every ramified critical pair \(P\), the ratios \(R_P(\theta_\iota^n, 1)\) are nonzero and the family \(\iota \mapsto R_P(\theta_\iota^n, 1)^{-1}\) has height at most \(x_1\).  Then the families \(\iota \mapsto \bigl(\prod_{1 \le i < j \le m}\rho_{ij}(\theta_\iota)\bigr)^{-1}\) and \(\iota \mapsto \bigl(\prod_{0 \le i, j < m}\rho'_{ij}(\theta_\iota)\bigr)^{-1}\) have height at most \(m^2tx_1\); the first is a product of \(m(m-1)/2\) inverse ratios \(\rho_{ij}^{-1}\), the second of \(m^2\).
\end{problem}

\section*{Hints}

\subsection*{\S\ref{sec:volumes}}
\begin{description}
\item[\ref{pr:gram}] \(\langle u e_a, u e_b\rangle = \langle e_a, u^*u\, e_b\rangle\), and \(u^*u\) is positive semidefinite with kernel \(\ker u\).
\item[\ref{pr:scalar-gram}] Rewrite the hypothesis as an identity of endomorphisms of \(U\).
\item[\ref{pr:square-chain}] In orthonormal bases of spaces of equal dimension, \(Z(u) = \lvert\det u\rvert^2\).
\item[\ref{pr:position}] Problem~\ref{pr:gram} with \(U = \C\).
\item[\ref{pr:smallest}] Problem~\ref{pr:gram} on a line.
\end{description}

\subsection*{\S\ref{sec:one-lift}}
\begin{description}
\item[\ref{pr:one-lift}] Problem~\ref{pr:scalar-gram} on \(H_g(W_j)\), which \(\Rc_j(\xi; W_j)\) preserves; then the equality of determinants and \(\dim H_g(W_j) = d_T\).  For positivity, Problem~\ref{pr:gram} and the injectivity of \(\hat u_{\xi,j}\).
\end{description}

\subsection*{\S\ref{sec:twists}}
\begin{description}
\item[\ref{pr:powers}] The definition of \(\vartheta^n\) in Section~\ref{ob-sec:twists}, and Definition~\ref{c1-def:twist-factor}.
\item[\ref{pr:rho-one}] At the trivial twist \(Q_1 = \varnothing\) and \(1^{j-i} = 1\).
\end{description}

\subsection*{\S\ref{sec:factor}}
\begin{description}
\item[\ref{pr:abs-ratio}] Evaluate at \(1\) first.
\item[\ref{pr:chain-det}] For the first statement, Theorem~\ref{c4-thm:chain-gram} at \(\vartheta = 1\), and injectivity.  For the formula, Problems~\ref{pr:scalar-gram}, \ref{pr:powers} and~\ref{pr:one-lift}; compare the factors \(c_{Q}\) for \(\vartheta^j\) and \(\vartheta\).
\item[\ref{pr:quotient}] Take absolute values in Problem~\ref{pr:chain-det} and divide by the case \(\vartheta = 1\), using Problems~\ref{pr:abs-ratio} and~\ref{pr:rho-one}.  For \(T = 0\), Corollary~\ref{c2-cor:equal}.
\end{description}

\subsection*{\S\ref{sec:aux}}
\begin{description}
\item[\ref{pr:count-ratios}] For a product, add the upper bounds for the heights of its factors (Problem~\ref{c5-pr:height-mul}).  A height bound may be increased, and \(m(m-1)/2 \le m^2\).
\end{description}
